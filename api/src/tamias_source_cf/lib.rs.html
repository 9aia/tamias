3//! tamias-source-cf — the `cf:` ref kind + source: a **Cloudflare registry**.
4//! `<name>@<version>` resolves against the registry index (`<base>/index.json`),
5//! then the tarball is fetched over HTTP (delegating to `tamias-source-http`).
6//!
7//! The grammar is `cf:<account>/<bucket…>/<name>@<version>`, where the address
8//! derives the base URL: a full `http(s)://…` passes through, a first segment
9//! with no dot is a `workers.dev` subdomain, and later segments are a path
10//! prefix.
11

25/// A parsed `cf:` ref.
26#[derive(Clone, Debug, PartialEq, Eq)]
27pub struct CfSpec {
28    /// Registry base URL, derived from the address.
29    pub base_url: String,
30    /// The `<account>/<bucket…>` address as written (provenance).
31    pub address: String,
32    /// Bare package name.
33    pub name: String,
34    /// Declared version.
35    pub version: String,
36}

38impl RefKind for CfRefKind {
39    fn claims(&self, input: &str) -> bool {
40        input.starts_with("cf:")
41    }
42
43    fn parse(&self, input: &str) -> Result<Spec, Error> {
44        parse_cf(input.strip_prefix("cf:").unwrap_or_default())
45    }
46}

48/// `cf:<account>/<bucket…>/<name>@<version>` → a [`CfSpec`].
49fn parse_cf(body: &str) -> Result<Spec, Error> {
50    let bad = || {
51        Error::Failed(format!(
52            "bad cf ref '{body}' — expected cf:<account>/<bucket…>/<name>@<version>"
53        ))
54    };
55    let (addr, version) = body.rsplit_once('@').ok_or_else(bad)?;
56    let (address, name) = addr.rsplit_once('/').ok_or_else(bad)?;
57    if address.is_empty() || name.is_empty() || version.is_empty() {
58        return Err(bad());
59    }
60    Ok(Box::new(CfSpec {
61        base_url: cf_base_url(address),
62        address: address.to_string(),
63        name: name.to_string(),
64        version: version.to_string(),
65    }))
66}
67
68/// `<account>/<bucket…>` → a registry base URL. A full `http(s)://…` address
69/// passes through; a first segment with no dot is a `workers.dev` subdomain;
70/// later segments are a path prefix.
71fn cf_base_url(address: &str) -> String {
72    if address.starts_with("http://") || address.starts_with("https://") {
73        return address.trim_end_matches('/').to_string();
74    }
75    let mut segments = address.split('/');
76    let host = segments.next().unwrap_or_default();
77    let host = if host.contains('.') {
78        host.to_string()
79    } else {
80        format!("{host}.workers.dev")
81    };
82    let prefix: Vec<&str> = segments.collect();
83    if prefix.is_empty() {
84        format!("https://{host}")
85    } else {
86        format!("https://{host}/{}", prefix.join("/"))
87    }
88}

102struct CfPlan {
103    url: String,
104    /// The index's recorded `dir_digest` of the published payload root —
105    /// the *expected* unpacked-payload digest. The tarball itself carries
106    /// no pin, so this is the only provenance check between "the registry
107    /// said hello@1.0.0 lives at this URL" and "these bytes are it".
108    digest: String,
109}
110
111impl Plan for CfPlan {}
112
113impl Source for CfSource {
114    fn accepts(&self, spec: &dyn Any) -> bool {
115        spec.is::<CfSpec>()
116    }
117
118    fn plan(&self, spec: &dyn Any) -> Result<Box<dyn Plan>, Error> {
119        let CfSpec {
120            base_url,
121            name,
122            version,
123            ..
124        } = spec
125            .downcast_ref::<CfSpec>()
126            .ok_or_else(|| Error::Failed("CfSource only plans cf refs".to_string()))?;
127        let base = base_url.trim_end_matches('/').to_string();
128        let entries = HttpRegistry::from_env(base.clone()).entries()?;
129        let entry = entries
130            .into_iter()
131            .find(|e| &e.name == name && e.version.as_deref() == Some(version.as_str()))
132            .ok_or_else(|| Error::Failed(format!("{name}@{version} not found in {base}")))?;
133        Ok(Box::new(CfPlan {
134            url: format!("{base}/{}", entry.file),
135            digest: entry.digest,
136        }))
137    }
138
139    fn fetch(&self, plan: &dyn Plan, dest_dir: &Path) -> Result<Fetched, Error> {
140        // A `cf:` plan is an http tarball — the http source does the rest.
141        let CfPlan { url, digest } = plan
142            .downcast_ref::<CfPlan>()
143            .ok_or_else(|| Error::Failed("CfSource only fetches cf plans".to_string()))?;
144        let fetched = HttpSource::new().fetch(&HttpPlan::new(url.clone(), None), dest_dir)?;
145        // Verify the unpacked payload against the digest the registry
146        // index recorded at publish — a swapped or truncated tarball
147        // fails here instead of installing under the right name@version.
148        let actual = dir_digest(&payload_root(&fetched.dir)?)?;
149        if actual != *digest {
150            return Err(Error::Failed(format!(
151                "payload digest mismatch for {url}: {actual} ≠ index {digest}"
152            )));
153        }
154        Ok(fetched)
155    }
156}

169    /// A local registry serving `hello@0.1.0`; returns its base URL + the
170    /// kept-alive root dir.
171    fn serve_registry(port: &str) -> (String, tempfile::TempDir) {
172        use tamias_registry_local::{serve_http, LocalRegistry};
173        let reg = tempfile::tempdir().unwrap();
174        let root = reg.path().to_path_buf();
175        let src = tempfile::tempdir().unwrap();
176        std::fs::write(
177            src.path().join("tamias.yaml"),
178            "name: hello\nversion: 0.1.0\n",
179        )
180        .unwrap();
181        std::fs::write(src.path().join("a.txt"), "a").unwrap();
182        let manifest = tamias_core::Manifest::load(src.path()).unwrap();
183        LocalRegistry::new(&root)
184            .publish(src.path(), &manifest)
185            .unwrap();
186        let addr = format!("127.0.0.1:{port}");
187        let served = root.clone();
188        std::thread::spawn(move || {
189            let _ = serve_http(&addr, &served);
190        });
191        for _ in 0..100 {
192            if std::net::TcpStream::connect(format!("127.0.0.1:{port}")).is_ok() {
193                break;
194            }
195            std::thread::sleep(std::time::Duration::from_millis(100));
196        }
197        (format!("http://127.0.0.1:{port}"), reg)
198    }

200    #[test]
201    fn grammar_plan_fetch_and_errors() {
202        let refs = Refs::new().with(Arc::new(CfRefKind));
203        let sources = Sources::new().with(Arc::new(CfSource::new()));
204
205        // grammar + base-URL derivation
206        let spec = refs.parse("cf:9aia/bucket/hello@0.1.0").unwrap();
207        let CfSpec {
208            base_url,
209            address,
210            name,
211            version,
212        } = *spec.downcast::<CfSpec>().unwrap();
213        assert_eq!(base_url, "https://9aia.workers.dev/bucket");
214        assert_eq!(address, "9aia/bucket");
215        assert_eq!(name, "hello");
216        assert_eq!(version, "0.1.0");
217        let spec = refs.parse("cf:reg.example.com/hi@1.2.3").unwrap();
218        assert_eq!(
219            spec.downcast::<CfSpec>().unwrap().base_url,
220            "https://reg.example.com"
221        );
222        assert!(refs.parse("cf:").is_err());
223        assert!(refs.parse("cf:name@1.0.0").is_err());
224        assert!(refs.parse("cf:a/b/name").is_err());
225        assert!(refs.parse("cf:a/b/@1.0.0").is_err());
226        assert!(refs.parse("cf:a/b/name@").is_err());
227        assert!(CfRefKind.claims("cf:x/y@1"));
228        assert!(!CfRefKind.claims("./x"));
229
230        // live registry: plan + fetch
231        let (base, _reg) = serve_registry("8911");
232        let spec = refs.parse(&format!("cf:{base}/hello@0.1.0")).unwrap();
233        let planned = sources.plan(&*spec).unwrap();
234        let dest = tempfile::tempdir().unwrap();
235        let fetched = planned.fetch(dest.path()).unwrap();
236        assert!(fetched.resolved.starts_with("sha256:"));
237        assert!(dest.path().join("a.txt").is_file());
238
239        // missing version → plan error
240        let spec = refs.parse(&format!("cf:{base}/ghost@9.9.9")).unwrap();
241        assert!(sources.plan(&*spec).is_err());
242        // dead registry → plan error
243        let spec = refs.parse("cf:http://127.0.0.1:1/ghost@1.0.0").unwrap();
244        assert!(sources.plan(&*spec).is_err());
245
246        // foreign spec / plan
247        let source = CfSource::new();
248        assert!(!source.accepts(&1u8));
249        assert!(source.plan(&1u8).is_err());
250        assert!(source.fetch(&ForeignPlan, dest.path()).is_err());
251    }

253    #[test]
254    fn fetch_rejects_payload_digest_mismatch() {
255        let refs = Refs::new().with(Arc::new(CfRefKind));
256        let sources = Sources::new().with(Arc::new(CfSource::new()));
257        let (base, reg) = serve_registry("8912");
258        let spec = refs.parse(&format!("cf:{base}/hello@0.1.0")).unwrap();
259        let planned = sources.plan(&*spec).unwrap();
260        // happy path first — the index digest verifies the unpacked payload
261        let d1 = tempfile::tempdir().unwrap();
262        assert!(planned.fetch(d1.path()).is_ok());
263        // swap the served tarball for a different payload under the same
264        // name@version → the index digest must reject it
265        let evil = tempfile::tempdir().unwrap();
266        std::fs::write(
267            evil.path().join("tamias.yaml"),
268            "name: hello\nversion: 0.1.0\n",
269        )
270        .unwrap();
271        std::fs::write(evil.path().join("a.txt"), "evil").unwrap();
272        tamias_core::pack_tgz_atomic(evil.path(), &reg.path().join("hello/0.1.0.tgz")).unwrap();
273        let d2 = tempfile::tempdir().unwrap();
274        let err = planned.fetch(d2.path()).unwrap_err();
275        assert!(err.to_string().contains("digest mismatch"), "{err}");
276    }
