3//! Config port (`tamias-config`) — the tamias config model + format port.
4//!

58/// The tamias config.

73/// A config file format — parse text → [`Config`], and render back.
74pub trait ConfigFormat: Send + Sync {
75    /// The file extensions this format claims (no leading dot).
76    fn extensions(&self) -> &'static [&'static str];
77
78    /// Parse `text`.
79    ///
80    /// # Errors
81    /// Fails on a syntax or type error.
82    fn parse(&self, text: &str) -> Result<Config, Error>;
83
84    /// Render `config` back to this format.
85    ///
86    /// # Errors
87    /// Fails on a serialize error.
88    fn render(&self, config: &Config) -> Result<String, Error>;
89}
90
91/// The format that claims `path`'s extension, if any.
92#[must_use]
93pub fn format_for<'a>(
94    path: &Path,
95    formats: &'a [&'a dyn ConfigFormat],
96) -> Option<&'a dyn ConfigFormat> {
97    let ext = path.extension().and_then(|e| e.to_str())?;
98    formats
99        .iter()
100        .copied()
101        .find(|f| f.extensions().contains(&ext))
102}
103
104/// Load a specific config file, parsing with the format that claims its
105/// extension.
106///
107/// # Errors
108/// Fails on a read/parse error or an unclaimed extension.
109pub fn load_file(path: &Path, formats: &[&dyn ConfigFormat]) -> Result<Config, Error> {
110    let format = format_for(path, formats)
111        .ok_or_else(|| Error::Failed(format!("no config format for {}", path.display())))?;
112    let text =
113        std::fs::read_to_string(path).map_err(Error::io(format!("read {}", path.display())))?;
114    format.parse(&text)
115}
116
117/// Load the first config file found in `dir` (YAML first — [`CONFIG_NAMES`]).
118///
119/// # Errors
120/// Fails on a read/parse error; `Ok(None)` when no config file exists.
121pub fn load_in(
122    dir: &Path,
123    formats: &[&dyn ConfigFormat],
124) -> Result<Option<(PathBuf, Config)>, Error> {
125    for name in CONFIG_NAMES {
126        let path = dir.join(name);
127        if !path.is_file() {
128            continue;
129        }
130        let config = load_file(&path, formats)?;
131        return Ok(Some((path, config)));
132    }
133    Ok(None)
134}

136#[cfg(test)]
137#[cfg_attr(coverage, coverage(off))]
138mod tests {
139    use super::*;
140
141    /// A toy format claiming `yml`, parsing `"ok"` to the default.
142    struct Toy;
143    impl ConfigFormat for Toy {
144        fn extensions(&self) -> &'static [&'static str] {
145            &["yml"]
146        }
147        fn parse(&self, text: &str) -> Result<Config, Error> {
148            if text.trim() == "ok" {
149                Ok(Config::default())
150            } else {
151                Err(Error::Failed("bad toy".into()))
152            }
153        }
154        fn render(&self, _config: &Config) -> Result<String, Error> {
155            Ok("ok".into())
156        }
157    }
158
159    #[test]
160    fn format_for_load_file_and_load_in() {
161        let toy = Toy;
162        let formats: [&dyn ConfigFormat; 1] = [&toy];
163        assert!(format_for(Path::new("a.yml"), &formats).is_some());
164        assert!(format_for(Path::new("a.json"), &formats).is_none());
165        assert!(format_for(Path::new("noext"), &formats).is_none());
166
167        let dir = tempfile::tempdir().unwrap();
168        assert!(load_in(dir.path(), &formats).unwrap().is_none());
169
170        let p = dir.path().join("tamias.yml");
171        std::fs::write(&p, "ok").unwrap();
172        let (found, cfg) = load_in(dir.path(), &formats).unwrap().unwrap();
173        assert_eq!(found, p);
174        assert_eq!(cfg, Config::default());
175
176        assert!(load_file(&dir.path().join("x.json"), &formats).is_err());
177        assert!(load_file(&dir.path().join("gone.yml"), &formats).is_err());
178        std::fs::write(&p, "nope").unwrap();
179        assert!(load_in(dir.path(), &formats).is_err());
180
181        assert_eq!(toy.render(&Config::default()).unwrap(), "ok");
182        let mut c = Config::default();
183        c.store.kind = Some("local".into());
184        c.publish.default = Some("r".into());
185        c.registries.insert(
186            "r".into(),
187            Registry {
188                kind: Some("cf".into()),
189                url: Some("u".into()),
190                token_env: Some("T".into()),
191            },
192        );
193        assert_eq!(c.registries["r"].url.as_deref(), Some("u"));
194    }
