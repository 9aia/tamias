1//! entity_worker — the `ci-job` consumption loop (P-002 §4.1): the
2//! entity-plane replacement for the redis `WorkQueue` pull path.
3//!
4//! One worker process polls `EntityApi::list_entities("ci-job")`,
5//! filters pending jobs this box can run (known key + `runsOn` ⊆
6//! `[worker] labels`), CAS-claims via `ci.claim`, executes through
7//! the same `run_job` pipeline every other host uses (worktree →
8//! runner → status sink), renews the lease on `run_job`'s ticker,
9//! and settles with `ci.complete`. Claim loss (409 fence mismatch)
10//! abandons the run — a peer owns it; a `cancelRequested` flag on
11//! the entity kills the child and settles `cancelled`.
12//!
13//! The loop is split into a narrow synchronous [`JobApi`] port (so
14//! the state machine unit-tests against an in-memory fake) and
15//! [`EntityJobApi`], the `hypha_node_api::EntityApi` adapter that
16//! bridges the async trait onto a current-thread tokio runtime —
17//! `HttpEntityApi` is blocking-ureq inside `spawn_blocking`, which
18//! needs a runtime context bupha otherwise doesn't carry.
19

41/// Default claim lease — the plane's `DEFAULT_LEASE_SECS` (5 min,
42/// renewed ~every 2 min by `run_job`'s ticker).
43const DEFAULT_LEASE_SECS: u64 = 300;
44
45/// `BUPHA_ENTITY_LEASE_SECS` overrides the claim lease — same env
46/// override style as `BUPHA_RENEW_TICKS`/`BUPHA_WORKER_POLL_MS`.
47fn lease_secs() -> u64 {
48    std::env::var("BUPHA_ENTITY_LEASE_SECS")
49        .ok()
50        .and_then(|v| v.parse::<u64>().ok())
51        .filter(|n| *n > 0)
52        .unwrap_or(DEFAULT_LEASE_SECS)
53}
54
55/// Idle poll between scans — `BUPHA_WORKER_POLL_MS` (shared with the
56/// redis `bupha worker` cadence), default 1s.
57fn poll_ms() -> u64 {
58    std::env::var("BUPHA_WORKER_POLL_MS")
59        .ok()
60        .and_then(|v| v.parse::<u64>().ok())
61        .unwrap_or(1000)
62}

66/// A `ci-job` entity as the worker sees it — the wire id (ops target
67/// it verbatim) plus the parsed indexDoc.
68#[derive(Debug, Clone)]
69pub struct EntityJob {
70    /// `ci:{key}:{sha}:{attempt}` — contains `/` via the repo part;
71    /// `HttpEntityApi` percent-encodes it into the path.
72    pub id: String,
73    /// The parsed + validated indexDoc (`CiJob::from_index_doc` —
74    /// the same trust boundary the plane applies at create).
75    pub job: CiJob,
76}
77
78/// A won `ci.claim` — every holder op (`renew`/`release`/`complete`)
79/// must echo `worker` + `fence`; a mismatch is a 409 = claim lost.
80#[derive(Debug, Clone)]
81pub struct HeldClaim {
82    pub id: String,
83    pub worker: String,
84    pub fence: u64,
85}
86
87/// `JobApi::claim` resolution — `Lost` covers every "someone else
88/// owns it" shape (409 heldBy, label-gate 409, entity gone); the
89/// caller rescans rather than distinguishing.
90#[derive(Debug)]
91pub enum ClaimOutcome {
92    Won(HeldClaim),
93    Lost,
94}
95
96/// The narrow synchronous port the claim loop speaks. [`EntityJobApi`]
97/// adapts `hypha_node_api::EntityApi` (async, ureq-backed);
98/// tests substitute an in-memory fake.
99pub trait JobApi: Send + Sync {
100    /// `list_entities("ci-job")` → parsed pending-capable docs.
101    fn list_jobs(&self) -> io::Result<Vec<EntityJob>>;
102    /// `GET /api/entities/{id}` → the current doc (cancel/holder
103    /// checks). `None` = unknown id.
104    fn get_job(&self, id: &str) -> io::Result<Option<EntityJob>>;
105    /// `ci.claim {worker, leaseSecs}` — CAS `pending→running`.
106    fn claim(&self, id: &str, worker: &str, lease_secs: u64) -> io::Result<ClaimOutcome>;
107    /// `ci.renew {worker, fence, leaseSecs}` — `false` = fence
108    /// mismatch (claim lost) OR `cancelRequested` observed; the
109    /// caller disambiguates via `get_job` after the run aborts.
110    fn renew(&self, claim: &HeldClaim, lease_secs: u64) -> io::Result<bool>;
111    /// `ci.complete {worker, fence, result, description?}` — terminal
112    /// transition. `Err` on transport failure or 409 (fence lost —
113    /// the peer that re-claimed owns the settle).
114    fn complete(
115        &self,
116        claim: &HeldClaim,
117        result: &str,
118        description: Option<&str>,
119    ) -> io::Result<()>;
120    /// `ci.release {worker, fence, reason?}` — graceful
121    /// `running→pending` (shutdown path).
122    fn release(&self, claim: &HeldClaim, reason: &str) -> io::Result<()>;
123}

125// --- EntityApi adapter ---------------------------------------------------

127/// `JobApi` over `hypha_node_api::EntityApi` — the production impl
128/// (`HttpEntityApi` against a hypha/peregrin node; any other
129/// `EntityApi` impl works too, e.g. a hub's `SyncEntityApi`).
130///
131/// The trait is `async_trait` and `HttpEntityApi` hops onto
132/// `spawn_blocking`, so calls need a tokio runtime — this type owns
133/// a current-thread one and `block_on`s each call. bupha stays
134/// sync/no-runtime everywhere else; the runtime is private to the
135/// worker.
136pub struct EntityJobApi {
137    api: Arc<dyn EntityApi>,
138    rt: tokio::runtime::Runtime,
139}
140
141impl EntityJobApi {
142    /// Wrap any `EntityApi` impl.
143    ///
144    /// # Errors
145    /// Fails if the current-thread runtime can't be built.
146    pub fn new(api: Arc<dyn EntityApi>) -> io::Result<Self> {
147        let rt = tokio::runtime::Builder::new_current_thread()
148            .enable_all()
149            .build()
150            .map_err(io::Error::other)?;
151        Ok(Self { api, rt })
152    }
153
154    /// Convenience for the common case — ureq `HttpEntityApi`
155    /// against `base` (e.g. `http://127.0.0.1:18787`) with an
156    /// optional bearer token (paired node credential).
157    ///
158    /// # Errors
159    /// Fails if the runtime can't be built.
160    pub fn http(base: &str, token: Option<String>) -> io::Result<Self> {
161        Self::new(Arc::new(hypha_node_api::api::HttpEntityApi::new(
162            base, token,
163        )))
164    }
165
166    fn submit(&self, id: &str, kind: &str, payload: serde_json::Value) -> io::Result<()> {
167        let api = Arc::clone(&self.api);
168        let id = id.to_string();
169        let op = EntityOp {
170            kind: kind.to_string(),
171            payload,
172        };
173        self.rt
174            .block_on(async move { api.submit_op(&id, &op).await })
175            .map_err(io::Error::other)
176    }
177
178    /// `submit_op`'s error channel is a formatted string
179    /// (`"submit op: HTTP 409 — {body}"`) — the heldBy detail rides
180    /// in the body; the loop only needs the status class.
181    fn is_conflict(e: &io::Error) -> bool {
182        let msg = e.to_string();
183        msg.contains("HTTP 409") || msg.contains("HTTP 404")
184    }
185}
186
187/// `EntityDto.index_doc` → `EntityJob` — the same validation the
188/// plane applies at create; hostile docs drop out of the candidate
189/// set rather than reaching git argv.
190fn parse_job(dto: &EntityDto) -> Option<EntityJob> {
191    if dto.kind != CI_JOB_KIND {
192        return None;
193    }
194    let now = chrono::Utc::now().to_rfc3339();
195    let mut job = CiJob::from_index_doc(&dto.index_doc, &now).ok()?;
196    job.id = dto.id.clone();
197    Some(EntityJob {
198        id: dto.id.clone(),
199        job,
200    })
201}
202
203impl JobApi for EntityJobApi {
204    fn list_jobs(&self) -> io::Result<Vec<EntityJob>> {
205        let api = Arc::clone(&self.api);
206        let dtos = self
207            .rt
208            .block_on(api.list_entities(Some(CI_JOB_KIND)))
209            .map_err(io::Error::other)?;
210        Ok(dtos.iter().filter_map(parse_job).collect())
211    }
212
213    fn get_job(&self, id: &str) -> io::Result<Option<EntityJob>> {
214        let api = Arc::clone(&self.api);
215        let id = id.to_string();
216        match self.rt.block_on(api.entity(&id)) {
217            Ok(dto) => Ok(parse_job(&dto)),
218            // 404 = gone — not an error for the poll/claim paths.
219            Err(e) if e.contains("HTTP 404") => Ok(None),
220            Err(e) => Err(io::Error::other(e)),
221        }
222    }
223
224    fn claim(&self, id: &str, worker: &str, lease_secs: u64) -> io::Result<ClaimOutcome> {
225        let payload = json!({ "worker": worker, "leaseSecs": lease_secs });
226        match self.submit(id, "ci.claim", payload) {
227            Ok(()) => {
228                // `submit_op` discards the op's response body, so the
229                // fence comes from the entity doc the claim just
230                // wrote — ops serialize node-side, the re-read is
231                // consistent.
232                match self.get_job(id)? {
233                    Some(ej)
234                        if ej.job.state == JobState::Running
235                            && ej.job.claimed_by.as_deref() == Some(worker) =>
236                    {
237                        Ok(ClaimOutcome::Won(HeldClaim {
238                            id: id.to_string(),
239                            worker: worker.to_string(),
240                            fence: ej.job.fence,
241                        }))
242                    }
243                    // Claim accepted but the doc disagrees — treat as
244                    // lost rather than run unfenced.
245                    _ => Ok(ClaimOutcome::Lost),
246                }
247            }
248            Err(e) if Self::is_conflict(&e) => Ok(ClaimOutcome::Lost),
249            Err(e) => Err(e),
250        }
251    }
252
253    fn renew(&self, claim: &HeldClaim, lease_secs: u64) -> io::Result<bool> {
254        let payload = json!({
255            "worker": claim.worker,
256            "fence": claim.fence,
257            "leaseSecs": lease_secs,
258        });
259        match self.submit(&claim.id, "ci.renew", payload) {
260            Ok(()) => {
261                // The 200 proves holder+fence; the extra read is for
262                // `cancelRequested` — the worker's kill switch (the
263                // renew loop is the only place a running job can
264                // observe it, P-002 §2 `ci.cancel`).
265                match self.get_job(&claim.id)? {
266                    Some(ej) => Ok(ej.job.state == JobState::Running
267                        && ej.job.claimed_by.as_deref() == Some(claim.worker.as_str())
268                        && !ej.job.cancel_requested),
269                    None => Ok(false),
270                }
271            }
272            Err(e) if Self::is_conflict(&e) => Ok(false),
273            Err(e) => Err(e),
274        }
275    }
276
277    fn complete(
278        &self,
279        claim: &HeldClaim,
280        result: &str,
281        description: Option<&str>,
282    ) -> io::Result<()> {
283        let mut payload = json!({
284            "worker": claim.worker,
285            "fence": claim.fence,
286            "result": result,
287        });
288        if let Some(d) = description {
289            payload["description"] = json!(d);
290        }
291        self.submit(&claim.id, "ci.complete", payload)
292    }
293
294    fn release(&self, claim: &HeldClaim, reason: &str) -> io::Result<()> {
295        let payload = json!({
296            "worker": claim.worker,
297            "fence": claim.fence,
298            "reason": reason,
299        });
300        self.submit(&claim.id, "ci.release", payload)
301    }
302}

306/// `TryClaim` over a held entity claim — `run_job`'s renew ticker
307/// speaks this trait; every call maps onto `ci.renew`. `try_claim`/
308/// `is_done`/`mark_done` are single-box no-ops here: the entity CAS
309/// already happened and there is no peer-done set to consult.
310pub struct EntityClaim<'a> {
311    api: &'a dyn JobApi,
312    held: HeldClaim,
313    lease_secs: u64,
314}
315
316impl<'a> EntityClaim<'a> {
317    fn new(api: &'a dyn JobApi, held: HeldClaim, lease_secs: u64) -> Self {
318        Self {
319            api,
320            held,
321            lease_secs,
322        }
323    }
324}
325
326impl TryClaim for EntityClaim<'_> {
327    fn try_claim(&self, _repo: &str, _sha: &str, _worker: &str) -> io::Result<Option<Fence>> {
328        // The claim was already won via `ci.claim` — hand back its
329        // fence so callers that inspect it see the real token.
330        Ok(Some(Fence(self.held.fence)))
331    }
332
333    fn renew(&self, _repo: &str, _sha: &str, _worker: &str) -> io::Result<bool> {
334        self.api.renew(&self.held, self.lease_secs)
335    }
336}

340/// How one claimed job ended. `Skip` = not runnable after all
341/// (unknown key drift, hostile sha) — the loop releases the claim
342/// instead of settling a fake result.
343#[derive(Debug, Clone, Copy, PartialEq, Eq)]
344pub enum ExecEnd {
345    /// Run completed locally; `false` = failed.
346    Finished(bool),
347    /// Lease lost or cancel observed mid-run — `run_job` killed the
348    /// child; the loop decides complete-vs-abandon from the entity.
349    ClaimLost,
350    /// Release the claim untouched.
351    Skip,
352}
353
354/// The execution half of the loop — owns runner/sink/worktree
355/// plumbing. [`WatchExec`] is the production impl (the same
356/// `bupha-watch` builders `bupha worker`/`watch` use); tests
357/// substitute a scripted fake.
358pub trait JobExec: Send + Sync {
359    /// This worker's claim identity (`<host>:<pid>`) — goes into
360    /// `ci.claim`'s `worker` field.
361    fn worker(&self) -> &str;
362    /// Run one claimed job end-to-end. `claim` renews the entity
363    /// lease on `run_job`'s ticker.
364    fn exec(&self, job: &EntityJob, claim: &EntityClaim<'_>) -> ExecEnd;
365}

367/// Production executor — replicates `bupha worker`'s wiring (the
368/// CLI's `cmd_worker`): per-key status sinks, per-key runners,
369/// telemetry/attest/audit, one shared child slot (one build at a
370/// time per box). Queue persistence does NOT apply — entity state
371/// *is* the queue.
372pub struct WatchExec {
373    cfg: Arc<Config>,
374    watched: Arc<Vec<WatchedRepo>>,
375    sinks: std::collections::HashMap<String, Arc<dyn StatusSink>>,
376    runners: std::collections::HashMap<String, Arc<dyn Runner>>,
377    telemetry: Arc<dyn Telemetry>,
378    attest: Option<Arc<dyn Attest>>,
379    shared: Mutex<Shared>,
380    child: ChildSlot,
381    worker: String,
382}
383
384impl WatchExec {
385    /// Wire the executor against `cfg` — same fail-fast config
386    /// errors `bupha worker`/`watch` raise.
387    ///
388    /// # Panics
389    /// Exits the process on an unloadable config or unknown runner/
390    /// telemetry/audit kind — identical to the CLI paths.
391    #[must_use]
392    pub fn new(cfg: Arc<Config>) -> Self {
393        let token = secret(&cfg.runner.token_env).unwrap_or_default();
394        let watched: Vec<WatchedRepo> = cfg.watched().unwrap_or_else(|e| {
395            eprintln!("config: {e}");
396            std::process::exit(2);
397        });
398        let watched = Arc::new(watched);
399        let audit = build_audit(&cfg);
400        let sinks = build_sinks(&cfg, &watched, &token, audit.as_ref());
401        let runners = build_runners(&cfg, &watched);
402        let telemetry = build_telemetry(&cfg);
403        let attest = build_attest(&cfg);
404        let _ = std::fs::create_dir_all(&cfg.worktree.base);
405        Self {
406            cfg,
407            watched,
408            sinks,
409            runners,
410            telemetry,
411            attest,
412            shared: Mutex::new(Shared::default()),
413            child: Arc::new(Mutex::new(None)),
414            worker: worker_id(),
415        }
416    }
417}

419impl JobExec for WatchExec {
420    fn worker(&self) -> &str {
421        &self.worker
422    }
423
424    fn exec(&self, ej: &EntityJob, claim: &EntityClaim<'_>) -> ExecEnd {
425        let Some(repo) = self.watched.iter().find(|r| r.key == ej.job.key) else {
426            eprintln!("ci-job {}: unknown key {:?} — releasing", ej.id, ej.job.key);
427            return ExecEnd::Skip;
428        };
429        let sha = ej.job.sha.as_str();
430        // Last-line validation before the sha names worktree dirs and
431        // reaches git argv — the plane validated at create, but the
432        // wire is a trust boundary at every hop.
433        if !is_valid_sha(sha) {
434            eprintln!("ci-job {}: refusing unsafe sha {sha:?}", ej.id);
435            return ExecEnd::Skip;
436        }
437        let (Some(sink), Some(runner)) = (self.sinks.get(&repo.key), self.runners.get(&repo.key))
438        else {
439            eprintln!("ci-job {}: no sink/runner for key {:?}", ej.id, repo.key);
440            return ExecEnd::Skip;
441        };
442        self.telemetry.event(&RunEvent::Started {
443            repo: repo.key.clone(),
444            sha: sha.to_string(),
445        });
446        let t0 = std::time::Instant::now();
447        let end = run_job(
448            &self.cfg,
449            repo,
450            &**sink,
451            &**runner,
452            &self.shared,
453            &Worker {
454                child: &self.child,
455                claim,
456                id: &self.worker,
457            },
458            sha,
459        );
460        let wall = t0.elapsed();
461        settle_run(
462            &self.cfg,
463            &*self.telemetry,
464            &self.attest,
465            runner.name(),
466            &repo.key,
467            sha,
468            &end,
469            wall,
470        );
471        match end {
472            JobEnd::Finished(ok) => ExecEnd::Finished(ok),
473            JobEnd::ClaimLost => ExecEnd::ClaimLost,
474        }
475    }
476}

480/// What one keys this worker may claim: known job keys + the label
481/// set `[worker] labels` advertises. Built once from the config —
482/// `runsOn` matching is the same set-equality the plane re-checks
483/// server-side (P-002 §4: the caller pre-filters, the CAS makes the
484/// pick race-safe).
485#[derive(Debug, Default)]
486pub struct ClaimFilter {
487    pub keys: HashSet<String>,
488    pub labels: Vec<String>,
489}
490
491impl ClaimFilter {
492    /// From a loaded config — `cfg.watched()` keys + worker labels.
493    ///
494    /// # Errors
495    /// Propagates `Config::watched` validation failures.
496    pub fn from_config(cfg: &Config) -> Result<Self, String> {
497        Ok(Self {
498            keys: cfg.watched()?.iter().map(|w| w.key.clone()).collect(),
499            labels: cfg.worker.labels.clone(),
500        })
501    }
502}
503
504/// Pending candidates in claim order — oldest `enqueuedAt` first
505/// (`all`-lane FIFO; `latest` lanes hold ≤1 pending by construction,
506/// so the sort is a no-op there). Drops terminal/running/foreign/
507/// label-mismatched/hostile docs.
508#[must_use]
509pub fn candidates(jobs: &[EntityJob], filter: &ClaimFilter) -> Vec<EntityJob> {
510    let mut v: Vec<EntityJob> = jobs
511        .iter()
512        .filter(|ej| {
513            ej.job.state == JobState::Pending
514                && filter.keys.contains(&ej.job.key)
515                && ej.job.runs_on.iter().all(|l| filter.labels.contains(l))
516                && is_valid_sha(&ej.job.sha)
517        })
518        .cloned()
519        .collect();
520    v.sort_by(|a, b| {
521        a.job
522            .enqueued_at
523            .cmp(&b.job.enqueued_at)
524            .then_with(|| a.id.cmp(&b.id))
525    });
526    v
527}

529/// Outcome of one poll→claim→run→settle cycle.
530#[derive(Debug, Clone, Copy, PartialEq, Eq)]
531pub enum Step {
532    /// Shutdown requested — the loop exits.
533    Shutdown,
534    /// Nothing claimable this scan (or a transient api error).
535    Idle,
536    /// A job was claimed and run to its end.
537    Ran,
538}
539
540/// Settle one won claim — the `ci.complete`/`ci.release`/abandon
541/// decision tree (P-002 §2 semantics over the entity doc).
542fn settle(api: &dyn JobApi, exec: &dyn JobExec, ej: &EntityJob, held: HeldClaim) {
543    let lease = lease_secs();
544    let claim = EntityClaim::new(api, held.clone(), lease);
545    let end = exec.exec(ej, &claim);
546    match end {
547        ExecEnd::Skip => {
548            if let Err(e) = api.release(&held, "not runnable on this worker") {
549                eprintln!("ci.release {}: {e}", held.id);
550            }
551        }
552        ExecEnd::Finished(ok) => {
553            // A cancel that landed after the last renew tick still
554            // wins — we are still the holder, so `cancelled` is the
555            // honest result (the plane keeps `cancelRequested` until
556            // someone settles).
557            let cancelled = api
558                .get_job(&held.id)
559                .ok()
560                .flatten()
561                .is_some_and(|e| e.job.cancel_requested);
562            let (result, desc) = if cancelled {
563                ("cancelled", "ci cancelled")
564            } else if ok {
565                ("success", "ci passed")
566            } else {
567                ("failure", "ci failed")
568            };
569            if let Err(e) = api.complete(&held, result, Some(desc)) {
570                // 409 = fence lost mid/post-run — a peer owns the
571                // settle now; anything else is a transport problem.
572                // Either way there is nothing safe to retry here.
573                eprintln!("ci.complete {}: {e} — abandoning", held.id);
574            }
575        }
576        ExecEnd::ClaimLost => {
577            // renew returned false for one of two reasons: a real
578            // fence loss (peer re-claimed — abandon silently) or
579            // `cancelRequested` (we are still the holder — settle
580            // `cancelled`). The entity doc disambiguates.
581            let cancel_pending = matches!(
582                api.get_job(&held.id),
583                Ok(Some(e))
584                    if e.job.state == JobState::Running
585                        && e.job.claimed_by.as_deref() == Some(exec.worker())
586                        && e.job.cancel_requested
587            );
588            if cancel_pending {
589                if let Err(e) = api.complete(&held, "cancelled", Some("ci cancelled")) {
590                    eprintln!("ci.complete {}: {e} — claim lost, abandoning", held.id);
591                }
592            }
593        }
594    }
595}

597/// One poll→claim→run→settle cycle — exposed for tests; [`run`]
598/// loops it until `shutdown`.
599pub fn step(
600    filter: &ClaimFilter,
601    api: &dyn JobApi,
602    exec: &dyn JobExec,
603    shutdown: &dyn Fn() -> bool,
604) -> Step {
605    if shutdown() {
606        return Step::Shutdown;
607    }
608    let jobs = match api.list_jobs() {
609        Ok(j) => j,
610        Err(e) => {
611            eprintln!("ci-job poll: {e}");
612            return Step::Idle;
613        }
614    };
615    for cand in candidates(&jobs, filter) {
616        match api.claim(&cand.id, exec.worker(), lease_secs()) {
617            Ok(ClaimOutcome::Won(held)) => {
618                println!("ci-job claimed {} ({})", cand.id, held.worker);
619                settle(api, exec, &cand, held);
620                return Step::Ran;
621            }
622            // Lost the CAS — a peer owns it; rescan picks the next.
623            Ok(ClaimOutcome::Lost) => continue,
624            Err(e) => {
625                eprintln!("ci.claim {}: {e}", cand.id);
626                return Step::Idle;
627            }
628        }
629    }
630    Step::Idle
631}

633/// The worker loop — poll until `shutdown` (the e2e
634/// `BUPHA_WATCH_SHUTDOWN_FILE` hook in production; systemd's
635/// SIGTERM kills between jobs since runs are synchronous). Sleeps
636/// `poll` between scans — one job at a time per box, same as
637/// `bupha worker`.
638pub fn run(
639    filter: &ClaimFilter,
640    api: &dyn JobApi,
641    exec: &dyn JobExec,
642    shutdown: &dyn Fn() -> bool,
643) {
644    let poll = Duration::from_millis(poll_ms());
645    loop {
646        if step(filter, api, exec, shutdown) == Step::Shutdown {
647            break;
648        }
649        std::thread::sleep(poll);
650    }
651}

711    impl JobApi for FakeApi {
712        fn list_jobs(&self) -> io::Result<Vec<EntityJob>> {
713            Ok(self
714                .jobs
715                .lock()
716                .unwrap()
717                .values()
718                .map(|j| EntityJob {
719                    id: j.id.clone(),
720                    job: j.clone(),
721                })
722                .collect())
723        }
724
725        fn get_job(&self, id: &str) -> io::Result<Option<EntityJob>> {
726            Ok(self.get(id).map(|job| EntityJob {
727                id: job.id.clone(),
728                job,
729            }))
730        }
731
732        fn claim(&self, id: &str, worker: &str, _lease_secs: u64) -> io::Result<ClaimOutcome> {
733            let mut jobs = self.jobs.lock().unwrap();
734            let Some(job) = jobs.get_mut(id) else {
735                return Ok(ClaimOutcome::Lost);
736            };
737            if job.state != JobState::Pending {
738                return Ok(ClaimOutcome::Lost);
739            }
740            job.state = JobState::Running;
741            job.claimed_by = Some(worker.to_string());
742            job.fence += 1;
743            let fence = job.fence;
744            Ok(ClaimOutcome::Won(HeldClaim {
745                id: id.to_string(),
746                worker: worker.to_string(),
747                fence,
748            }))
749        }
750
751        fn renew(&self, claim: &HeldClaim, _lease_secs: u64) -> io::Result<bool> {
752            let mut jobs = self.jobs.lock().unwrap();
753            let Some(job) = jobs.get_mut(&claim.id) else {
754                return Ok(false);
755            };
756            if job.state != JobState::Running
757                || job.claimed_by.as_deref() != Some(claim.worker.as_str())
758                || job.fence != claim.fence
759                || job.cancel_requested
760            {
761                return Ok(false);
762            }
763            Ok(true)
764        }
765
766        fn complete(
767            &self,
768            claim: &HeldClaim,
769            result: &str,
770            _description: Option<&str>,
771        ) -> io::Result<()> {
772            if *self.fail_complete.lock().unwrap() {
773                return Err(io::Error::other("HTTP 409 — claim lost"));
774            }
775            let mut jobs = self.jobs.lock().unwrap();
776            let Some(job) = jobs.get_mut(&claim.id) else {
777                return Err(io::Error::other("HTTP 404"));
778            };
779            if job.state != JobState::Running
780                || job.claimed_by.as_deref() != Some(claim.worker.as_str())
781                || job.fence != claim.fence
782            {
783                return Err(io::Error::other("HTTP 409 — claim lost"));
784            }
785            job.state = JobState::from_result(result).unwrap();
786            job.result = Some(result.to_string());
787            Ok(())
788        }
789
790        fn release(&self, claim: &HeldClaim, _reason: &str) -> io::Result<()> {
791            let mut jobs = self.jobs.lock().unwrap();
792            if let Some(job) = jobs.get_mut(&claim.id) {
793                if job.claimed_by.as_deref() == Some(claim.worker.as_str())
794                    && job.fence == claim.fence
795                {
796                    job.state = JobState::Pending;
797                    job.claimed_by = None;
798                }
799            }
800            Ok(())
801        }
802    }

804    /// Scripted executor — records what it ran, returns a canned end
805    /// (or exercises the renew path when `renew_first` is set).
806    struct FakeExec {
807        worker: String,
808        end: ExecEnd,
809        renew_first: bool,
810        ran: Mutex<Vec<(String, String)>>,
811        renew_ok: Mutex<Option<bool>>,
812    }

814    impl FakeExec {
815        fn new(end: ExecEnd) -> Self {
816            Self {
817                worker: "testhost/1".into(),
818                end,
819                renew_first: false,
820                ran: Mutex::new(Vec::new()),
821                renew_ok: Mutex::new(None),
822            }
823        }
824    }

826    impl JobExec for FakeExec {
827        fn worker(&self) -> &str {
828            &self.worker
829        }
830
831        fn exec(&self, job: &EntityJob, claim: &EntityClaim<'_>) -> ExecEnd {
832            self.ran
833                .lock()
834                .unwrap()
835                .push((job.job.key.clone(), job.job.sha.clone()));
836            if self.renew_first {
837                // Exercise the TryClaim bridge mid-run — a false
838                // means cancel/loss and the run aborts.
839                let ok = claim
840                    .renew(&job.job.key, &job.job.sha, &self.worker)
841                    .unwrap();
842                *self.renew_ok.lock().unwrap() = Some(ok);
843                if !ok {
844                    return ExecEnd::ClaimLost;
845                }
846            }
847            self.end
848        }
849    }

917    fn cancel_during_run_settles_cancelled() {
918        let api = FakeApi::new();
919        api.insert(pending("o/n", "abc1234def56", 1));
920        // Executor renews mid-run; the api sees cancel_requested.
921        let exec = FakeExec {
922            renew_first: true,
923            ..FakeExec::new(ExecEnd::Finished(true))
924        };
925        // Pre-set cancel BEFORE exec — renew observes it → false →
926        // ClaimLost → settle path completes cancelled.
927        // (cancel_requested is set by ci.cancel; simulate directly.)
928        let filter = filter_for(&["o/n"]);
929        // Wrap: after claim, flip the flag — do it via a second api
930        // handle isn't possible mid-step, so set state through a
931        // custom exec that mutates via renew? Simplest: the fake's
932        // renew returns false when cancel_requested — set it by
933        // completing a manual claim first is complex; instead mark
934        // the job cancel_requested inside exec via a hook.
935        struct CancelExec<'a> {
936            inner: &'a FakeExec,
937            api: &'a FakeApi,
938        }
939        impl JobExec for CancelExec<'_> {
940            fn worker(&self) -> &str {
941                self.inner.worker()
942            }
943            fn exec(&self, job: &EntityJob, claim: &EntityClaim<'_>) -> ExecEnd {
944                // ci.cancel lands mid-run.
945                {
946                    let mut jobs = self.api.jobs.lock().unwrap();
947                    if let Some(j) = jobs.get_mut(&job.id) {
948                        j.cancel_requested = true;
949                    }
950                }
951                self.inner.exec(job, claim)
952            }
953        }
954        let cancel_exec = CancelExec {
955            inner: &exec,
956            api: &api,
957        };
958        assert_eq!(step(&filter, &api, &cancel_exec, &never), Step::Ran);
959        assert_eq!(*exec.renew_ok.lock().unwrap(), Some(false));
960        let job = api.get("ci:o/n:abc1234def56:1").unwrap();
961        assert_eq!(job.state, JobState::Cancelled);
962        assert_eq!(job.result.as_deref(), Some("cancelled"));
963    }
