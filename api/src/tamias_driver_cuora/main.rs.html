1//! tamias-driver-cuora — the pavo view driver for cuora, plus the
2//! engine ops surface.
3//!
4//! Verbs (the tamias driver contract):
5//!
6//! - `manifest` — print the [`DriverManifest`] JSON (discovery
7//!   probe; never touches vault state).
8//! - `capabilities` — `{"capabilities": ["pavo"], "verbs": […]}`.
9//! - `ops` — the engine ops manifest: every bounded `cuora` CLI
10//!   verb as a `cli.<verb>` wire query, derived from the engine's
11//!   own clap tree ([`cuora::command`]). Unbounded/stdin-bound
12//!   verbs (`serve`, `put`, `rotate`) are `surfaces`, never ops.
13//! - `run <argv>` — engine passthrough: `tamias-driver-cuora run
14//!   list` ≡ `cuora list`. Also the child side of `cli.*` wire
15//!   queries (`ops::run_verb` re-execs this same verb so engine
16//!   output never touches the serve stream).
17//! - `cli <cuora argv>` — the tamias `bins[]` shim verb: `tamias
18//!   bin exec` calls it with the standard `--dir`/`--name` driver
19//!   flags ahead of the user's argv; they're stripped and the rest
20//!   runs through `cuora::run_from` in-process with inherited
21//!   stdio — full engine parity (`serve` streams, `put` reads
22//!   stdin, globals/`--json`/`--version` all work).
23//! - `serve` — ndjson JSON-RPC on stdio: `{"op":"manifest"}`,
24//!   `{"op":"views"}`, `{"op":"view","view":…,"params":{…}}`,
25//!   `{"op":"query","query":…,"params":{…}}` →
26//!   `{"ok":true,"result":…}` / `{"ok":false,"error":…}`.
27//!   `query:"ops"` answers the ops manifest; `query:"cli.<verb>"`
28//!   runs the engine op.
29//!
30//! Optional surfaces: feature `mcp` embeds pavo-driver-kit's
31//! surface layer — `mcp` serves the ops manifest as MCP tools on
32//! stdio, equivalent to the standalone `tamias-driver-mcp` surface
33//! driver minus the PATH/discovery hop. (`cli` needs no feature:
34//! it's the bins-shim passthrough above, not a generated surface —
35//! the engine's own argv grammar is the contract.)
36//!
37//! Target resolution: each `view`/`query` builds `cuora_ops::Ops`
38//! from the request params — `config`/`env`/`actor` map to the
39//! CLI's `--config`/`--env`/`--actor`; absent params fall through
40//! to `$CUORA_CONFIG` → `./cuora.toml` → `~/.config/cuora/cuora.toml`,
41//! `$CUORA_ENV`, `$CUORA_ACTOR`/`$USER` — identical to `cuora`
42//! verbs. See `data.rs` for the secrets-safety contract: the view
43//! side renders vault *metadata* only, never values. Ops are the
44//! engine surface — `get`/`env` ops return values in captured
45//! stdout by design (the escape hatches); `put`/`rotate`/`serve`
46//! stay surfaces because they need stdin/never return.

58fn capabilities_json() -> Value {
59    let m = views::manifest();
60    // The kit's shape plus the view count every sibling driver
61    // advertises.
62    let mut v = kit::capabilities_value(&m);
63    v["views"] = json!(m.views.len());
64    v
65}

67/// Dispatch one parsed request to its response.
68fn dispatch(req: &Request) -> Response {
69    match req {
70        Request::Manifest => Response::ok_typed(&views::manifest()),
71        Request::Views => {
72            let summaries: Vec<ViewSummary> =
73                views::specs().iter().map(ViewSummary::from).collect();
74            Response::ok_typed(&summaries)
75        }
76        Request::View { view, params } => match views::view_result(view, params) {
77            Ok(result) => Response::ok_typed(&result),
78            Err(e) => Response::err(e),
79        },
80        Request::Render { kind, .. } => Response::err(format!(
81            "tamias-driver-cuora claims no renderers (kind '{kind}')"
82        )),
83        Request::Query { query, params } => {
84            // `ops` and `cli.<verb>` are engine ops; everything
85            // else is a view query — unknown names fail the op,
86            // degraded reads travel as `{"error": …}` data (the
87            // marker renderers understand), matching the reference
88            // driver's contract.
89            if query == "ops" {
90                Response::ok_typed(&wire::QueryResult {
91                    query: query.clone(),
92                    data: ops::manifest(),
93                })
94            } else if let Some(v) = ops::op_of(query) {
95                ops::run_verb(&v, params)
96            } else if views::known_query(query) {
97                let mut ctx = data::Ctx::open(params);
98                let payload = views::query_data(&mut ctx, query, params);
99                Response::ok_typed(&wire::QueryResult {
100                    query: query.clone(),
101                    data: payload,
102                })
103            } else {
104                Response::err(format!("unknown query: {query}"))
105            }
106        }
107        // The protocol grew `submit`/`jobs`/`notifications` ops after
108        // this driver was written; cuora mutations go through
109        // `cli.<verb>`/`run`, never `submit`, so they answer
110        // `ok:false` like any unsupported op.
111        Request::Submit { .. } | Request::Jobs | Request::Notifications => {
112            Response::err("tamias-driver-cuora implements no submit/jobs/notifications ops")
113        }
114    }
115}

117/// `cli` verb argv (after the `cli` token) → engine argv (argv[0]
118/// included). `tamias bin exec`/`driver exec` prepend the standard
119/// driver flags — `--dir <payload>` `--name <pkg>` — which the
120/// engine doesn't know; consume the leading run of those two
121/// (`--flag v` and `--flag=v` forms) and pass everything else
122/// through untouched. `--config` is deliberately NOT stripped: it's
123/// also a cuora global, so a user-passed `--config` must reach the
124/// engine's own config search.
125fn engine_argv(args: &[String]) -> Vec<String> {
126    let mut out = vec!["cuora".to_string()];
127    let mut i = 0;
128    while i < args.len() {
129        let a = args[i].as_str();
130        if a == "--dir" || a == "--name" {
131            // flag + separate value — skip both (a trailing flag
132            // with no value just ends the scan).
133            i = (i + 2).min(args.len());
134        } else if a.starts_with("--dir=") || a.starts_with("--name=") {
135            i += 1;
136        } else {
137            break;
138        }
139    }
140    out.extend(args[i..].iter().cloned());
141    out
142}

144fn serve() -> i32 {
145    // The kit owns the ndjson loop — stdin EOF and a dead client
146    // both end it quietly, and either way the driver exits 0.
147    let _ = kit::serve_stdio(dispatch);
148    0
149}

151fn main() {
152    let args: Vec<String> = std::env::args().skip(1).collect();
153    let code = match args.first().map(String::as_str) {
154        Some("manifest") => {
155            println!("{}", kit::manifest_json(&views::manifest()));
156            0
157        }
158        Some("capabilities") => {
159            println!(
160                "{}",
161                serde_json::to_string_pretty(&capabilities_json()).unwrap_or_default()
162            );
163            0
164        }
165        Some("ops") => {
166            println!("{}", ops::manifest());
167            0
168        }
169        // Engine passthrough — `tamias-driver-cuora run list` is
170        // `cuora list`. Also the child side of `cli.*` wire
171        // queries (ops::run_verb re-execs this same verb).
172        Some("run") => {
173            let argv = std::iter::once("cuora".to_string()).chain(args.iter().skip(1).cloned());
174            std::process::exit(cuora::run_from(argv));
175        }
176        // The tamias `bins[]` shim verb (P-023): `tamias bin exec`
177        // runs `<driver> cli --dir <payload> --name <pkg> <cuora
178        // argv>` for a declared `shim: cli` bin. cuora's engine IS
179        // the CLI, so this is a passthrough — strip the two standard
180        // driver flags and hand the rest to `cuora::run_from` with
181        // inherited stdio. That keeps full engine parity a generated
182        // ops→CLI surface can't express: globals before the verb
183        // (`--env`/`--config`/`--actor`), `--json` after it,
184        // `--version`/`--help`, the unbounded `serve` ndjson loop,
185        // and stdin-bound `put`/`rotate` all behave exactly like the
186        // retired `cuora` binary.
187        Some("cli") => {
188            let argv = engine_argv(&args[1..]);
189            std::process::exit(cuora::run_from(argv));
190        }
191        // Optional embedded surface (feature-gated): the
192        // pavo-driver-kit surface layer renders this driver's own
193        // ops manifest as MCP tools over stdio — the same ops the
194        // `cli.*` wire queries and the standalone
195        // tamias-driver-mcp binary expose.
196        #[cfg(feature = "mcp")]
197        Some("mcp") => {
198            let m = kit::surface::OpsManifest::from_value(&ops::manifest()).unwrap_or_default();
199            match kit::surface::mcp::serve_stdio(&m, &mut surface::Engine) {
200                Ok(()) => 0,
201                Err(e) => {
202                    eprintln!("tamias-driver-cuora mcp: {e}");
203                    1
204                }
205            }
206        }
207        Some("serve") | None => serve(),
208        _ => {
209            eprintln!(
210                "usage: tamias-driver-cuora manifest|capabilities|ops|run <argv>|cli <cuora argv>|serve\n       (feature `mcp` adds `mcp`)"
211            );
212            1
213        }
214    };
215    std::process::exit(code);
216}
