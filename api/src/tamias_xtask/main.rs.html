2//! xtask — repo chores. `cargo xtask <verb>` via the alias in
3//! `.cargo/config.toml`.
4//!
5//! - `check`    — `cargo fmt --check` + `cargo clippy -D warnings`
6//! - `test`     — `cargo nextest run` when installed, else `cargo test`
7//! - `coverage` — `cargo llvm-cov` (HTML → target/llvm-cov/html);
8//!   extra args pass through, e.g. `--fail-under-lines 97`
9//! - `install`  — release-install `tamias` into ~/.cargo/bin
10//! - `package`  — `cargo package --workspace` (verify: builds each packaged crate)
11//! - `smoke`    — install `tamias` from crates.io + run it (post-publish check)
12//! - `site`     — build the docs site (docs/ worktree, VitePress)
13
14use std::process::{Command, ExitStatus};
15
16fn run(args: &[&str]) -> ExitStatus {
17    let empty: &[&str] = &[];
18    let (prog, rest) = args.split_first().unwrap_or((&"true", empty));
19    Command::new(prog).args(rest).status().unwrap_or_else(|e| {
20        eprintln!("xtask: failed to spawn {prog}: {e}");
21        std::process::exit(1)
22    })
23}
24
25fn run_in(dir: &str, args: &[&str]) -> ExitStatus {
26    let empty: &[&str] = &[];
27    let (prog, rest) = args.split_first().unwrap_or((&"true", empty));
28    Command::new(prog)
29        .args(rest)
30        .current_dir(dir)
31        .status()
32        .unwrap_or_else(|e| {
33            eprintln!("xtask: failed to spawn {prog}: {e}");
34            std::process::exit(1)
35        })
36}

38fn have(tool: &str) -> bool {
39    Command::new(tool)
40        .arg("--version")
41        .stdout(std::process::Stdio::null())
42        .stderr(std::process::Stdio::null())
43        .status()
44        .map(|s| s.success())
45        .unwrap_or(false)
46}
47
48fn have_cargo_sub(sub: &str) -> bool {
49    Command::new("cargo")
50        .args([sub, "--version"])
51        .stdout(std::process::Stdio::null())
52        .stderr(std::process::Stdio::null())
53        .status()
54        .map(|s| s.success())
55        .unwrap_or(false)
56}

58fn check() -> i32 {
59    let ok = run(&["cargo", "fmt", "--all", "--", "--check"]).success()
60        && run(&[
61            "cargo",
62            "clippy",
63            "--workspace",
64            "--all-targets",
65            "--",
66            "-D",
67            "warnings",
68        ])
69        .success();
70    i32::from(!ok)
71}
72
73fn test() -> i32 {
74    let status = if have_cargo_sub("nextest") {
75        run(&["cargo", "nextest", "run", "--workspace"])
76    } else {
77        run(&["cargo", "test", "--workspace"])
78    };
79    status.code().unwrap_or(1)
80}
81
82fn coverage(extra: &[String]) -> i32 {
83    if !have_cargo_sub("llvm-cov") {
84        eprintln!("cargo-llvm-cov not installed — cargo install cargo-llvm-cov");
85        return 1;
86    }
87    let mut args = vec!["cargo", "llvm-cov"];
88    if have_cargo_sub("nextest") {
89        args.push("nextest");
90        args.extend(["--workspace", "--no-tests=pass"]);
91    } else {
92        args.push("--workspace");
93    }
94    if extra.is_empty() {
95        args.push("--html");
96    }
97    let owned: Vec<&str> = extra.iter().map(String::as_str).collect();
98    args.extend(owned);
99    run(&args).code().unwrap_or(1)
100}

102fn install() -> i32 {
103    run(&["cargo", "install", "--path", "crates/tamias-cli", "--force"])
104        .code()
105        .unwrap_or(1)
106}

108/// Package + *verify* every crate the way crates.io will see them — catches
109/// files that escape the crate (`include_str!("../../sibling/…")`) and any
110/// build that only works in-workspace. Run before a release.
111fn package() -> i32 {
112    run(&["cargo", "package", "--workspace", "--allow-dirty"])
113        .code()
114        .unwrap_or(1)
115}

117/// Registry-only smoke test: install `tamias` from crates.io into a temp root
118/// and run it. This is the **only** check that catches a *stale already-published
119/// dependency* — in-workspace builds hide it, and `cargo package --workspace`
120/// resolves workspace members by path, not by registry version.
121/// Run after publishing (see `scripts/release.sh`).
122fn smoke() -> i32 {
123    let root = std::env::temp_dir().join("tamias-smoke");
124    let _ = std::fs::remove_dir_all(&root);
125    let root_s = root.to_string_lossy().into_owned();
126    if run(&["cargo", "install", "tamias", "--root", &root_s, "--force"]).code() != Some(0) {
127        return 1;
128    }
129    let bin = root
130        .join("bin")
131        .join("tamias")
132        .to_string_lossy()
133        .into_owned();
134    run(&[&bin, "version"]).code().unwrap_or(1)
135}
136
137fn site() -> i32 {
138    if !std::path::Path::new("docs/package.json").exists() {
139        eprintln!("no docs/ worktree — `git worktree add --orphan -b docs docs` first");
140        return 1;
141    }
142    let pkg = if have("bun") { "bun" } else { "npm" };
143    let _ = run_in("docs", &[pkg, "install"]).success();
144    run_in("docs", &[pkg, "run", "docs:build"])
145        .code()
146        .unwrap_or(1)
147}

149fn main() {
150    let mut args = std::env::args().skip(1);
151    let cmd = args.next().unwrap_or_default();
152    let rest: Vec<String> = args.collect();
153    let code = match cmd.as_str() {
154        "check" => check(),
155        "test" => test(),
156        "coverage" => coverage(&rest),
157        "install" => install(),
158        "package" => package(),
159        "smoke" => smoke(),
160        "site" => site(),
161        other => {
162            eprintln!(
163                "usage: cargo xtask <check|test|coverage|install|package|smoke|site>\n  unknown verb: {other:?}"
164            );
165            2
166        }
167    };
168    std::process::exit(code);
169}
