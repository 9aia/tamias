2//! tamias-driver — the content **driver** host: discovery + dispatch of
3//! `tamias-driver-<type>` executables, with **no product knowledge** in
4//! tamias.
5//!
6//! A **driver** is an executable `tamias-driver-<type>` on `PATH` or in
7//! `<tamias-home>/drivers/`. tamias reads a package's `type:`, resolves its
8//! driver, and runs:
9//!
10//! ```text
11//! tamias-driver-<type> <verb> --dir <dir> --name <name> [--config <file>]
12//! ```
13//!
14//! `verb ∈ {install, deploy, uninstall}`; a driver may also answer
15//! `capabilities`. Installing a product drops its driver into
16//! `<tamias-home>/drivers/` (from the payload's `drivers/` dir), so tamias
17//! finds it. The engine never links a product crate.
18//!
19//! **Bundled drivers.** A payload's `drivers/` dir ships ready-to-run
20//! driver files:
21//!
22//! - `drivers/<name>` — portable drivers (interpreted scripts like
23//!   `#!/usr/bin/env python3` ship as source).
24//! - `drivers/bin/<target-triple>/<name>` — compiled drivers, one dir per
25//!   rustc target triple (`x86_64-unknown-linux-gnu`, …). At install time
26//!   only the dir matching [`target_triple`] is copied; a package that
27//!   ships platform dirs but none for this host fails the install with a
28//!   clear error rather than silently dropping the driver.
29//!
30//! Both forms land flat in `<tamias-home>/drivers/` (executable bit set),
31//! stay inside the payload digest, and are consent-gated — see
32//! [`drivers_allowed`].
33

39/// A driver verb (`install`/`deploy`/`uninstall`) can legitimately take a
40/// while — a hung one must not wedge the install or the single-threaded
41/// `tamias serve` loop.
42const DISPATCH_TIMEOUT: Duration = Duration::from_secs(120);
43
44/// `capabilities` is a discovery probe — seconds, not minutes.
45const CAPABILITIES_TIMEOUT: Duration = Duration::from_secs(10);
46
47/// The rustc target triple this build was compiled for
48/// (`x86_64-unknown-linux-gnu`) — captured at compile time by `build.rs`
49/// (`cargo:rustc-env`), so it always names a real cargo target even where
50/// `std::env::consts` can't reconstruct the vendor/os/env spelling.
51///
52/// This is the selector for a payload's `drivers/bin/<triple>/` dir —
53/// see [`register_bundled_drivers`].
54#[must_use]
55pub fn target_triple() -> &'static str {
56    env!("TAMIAS_TARGET_TRIPLE")
57}

59/// `<tamias-home>` — `$TAMIAS_HOME` → `$XDG_DATA_HOME/tamias` →
60/// `~/.local/share/tamias` → `./.tamias`.
61///
62/// The no-env fallback is `./.tamias` — the same project-local home the
63/// CLI's `store_root()`/`cache_root()`/`registry_root()` fall back to
64/// (`./.tamias/store`, `./.tamias/fetch`, `./.tamias/registry`), so the
65/// drivers dir lands at `./.tamias/drivers` beside them. (`./tamias`
66/// would collide with a `tamias` source checkout or package dir.)
67#[must_use]
68pub fn tamias_home() -> PathBuf {
69    std::env::var_os("TAMIAS_HOME")
70        .map(PathBuf::from)
71        .or_else(|| std::env::var_os("XDG_DATA_HOME").map(|d| PathBuf::from(d).join("tamias")))
72        .or_else(|| std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".local/share/tamias")))
73        .unwrap_or_else(|| PathBuf::from("./.tamias"))
74}

76/// The `<tamias-home>/drivers` dir.
77#[must_use]
78pub fn drivers_dir() -> PathBuf {
79    tamias_home().join("drivers")

82/// Find the driver for a content type — `tamias-driver-<kind>` on `PATH`,
83/// then `<tamias-home>/drivers/`.
84#[must_use]
85pub fn find_driver(kind: &str) -> Option<PathBuf> {
86    let name = format!("tamias-driver-{kind}");
87    if let Some(path) = lookup(&name) {
88        return Some(path);
89    }
90    // Otherwise map via capabilities: the first driver whose `types` include `kind`.
91    discovered_drivers()
92        .into_iter()
93        .find(|(_, path)| driver_handles(path, kind))
94        .map(|(_, path)| path)
95}
96
97/// `<name>` on `PATH`, then `<tamias-home>/drivers/`.
98fn lookup(name: &str) -> Option<PathBuf> {
99    let on_path = std::env::var_os("PATH").and_then(|path| {
100        std::env::split_paths(&path)
101            .map(|dir| dir.join(name))
102            .find(|candidate| candidate.is_file())
103    });
104    on_path.or_else(|| {
105        let candidate = drivers_dir().join(name);
106        candidate.is_file().then_some(candidate)
107    })
108}

110/// Whether `driver` reports handling `kind` (its `capabilities` `types`).
111fn driver_handles(driver: &Path, kind: &str) -> bool {
112    let mut cmd = std::process::Command::new(driver);
113    cmd.arg("capabilities");
114    let Ok(out) = output_with_timeout(&mut cmd, CAPABILITIES_TIMEOUT) else {
115        return false;
116    };
117    if !out.status.success() {
118        return false;
119    }
120    let Ok(value) = serde_json::from_slice::<serde_json::Value>(&out.stdout) else {
121        return false;
122    };
123    value
124        .get("types")
125        .and_then(|t| t.as_array())
126        .is_some_and(|a| a.iter().any(|x| x.as_str() == Some(kind)))
127}

129/// Every discovered driver as `(type, path)` — for `doctor`.
130#[must_use]
131pub fn discovered_drivers() -> Vec<(String, PathBuf)> {
132    let mut found: Vec<(String, PathBuf)> = Vec::new();
133    let mut scan = |dir: &Path| {
134        let Some(read) = std::fs::read_dir(dir).ok() else {
135            return;
136        };
137        for entry in read.flatten() {
138            let path = entry.path();
139            let kind = path
140                .file_name()
141                .and_then(|n| n.to_str())
142                .and_then(|n| n.strip_prefix("tamias-driver-"));
143            // `file_type` never follows symlinks — a linked driver isn't
144            // dispatchable (mirrors register_bundled_drivers).
145            let is_file = entry.file_type().is_ok_and(|t| t.is_file());
146            if let Some(kind) = kind {
147                if is_file && !found.iter().any(|(k, _)| k == kind) {
148                    found.push((kind.to_string(), path));
149                }
150            }
151        }
152    };
153    for dir in std::env::var_os("PATH")
154        .map(|p| std::env::split_paths(&p).collect::<Vec<_>>())
155        .unwrap_or_default()
156    {
157        scan(&dir);
158    }
159    scan(&drivers_dir());
160    found.sort();
161    found
162}

164/// Dispatch a verb to the content type's driver. Returns whether a driver
165/// ran (`false` = nothing to do — a `files` package, or no driver present).
166///
167/// # Errors
168/// Fails when a driver is found but exits non-zero.
169pub fn dispatch(
170    kind: &str,
171    verb: &str,
172    dir: &Path,
173    name: &str,
174    config: Option<&Path>,
175) -> Result<bool, Error> {
176    dispatch_with_timeout(kind, verb, dir, name, config, DISPATCH_TIMEOUT)
177}
178
179/// [`dispatch`] with an explicit bound — tests shrink it; production
180/// callers get [`DISPATCH_TIMEOUT`].
181fn dispatch_with_timeout(
182    kind: &str,
183    verb: &str,
184    dir: &Path,
185    name: &str,
186    config: Option<&Path>,
187    timeout: Duration,
188) -> Result<bool, Error> {
189    let Some(driver) = find_driver(kind) else {
190        return Ok(false);
191    };
192    let mut cmd = std::process::Command::new(&driver);
193    cmd.arg(verb).arg("--dir").arg(dir).arg("--name").arg(name);
194    if let Some(config) = config {
195        cmd.arg("--config").arg(config);
196    }
197    let out = output_with_timeout(&mut cmd, timeout)
198        .map_err(|e| Error::Failed(format!("run driver {}: {e}", driver.display())))?;
199    if !out.status.success() {
200        return Err(Error::Failed(format!(
201            "driver {} {verb} failed: {}",
202            driver.display(),
203            String::from_utf8_lossy(&out.stderr).trim()
204        )));
205    }
206    Ok(true)
207}

216/// Names of the driver files a payload ships under `drivers/` — regular
217/// files only, mirroring `register_bundled_drivers`'s filter, plus the
218/// platform bins under `drivers/bin/<triple>/` (reported as
219/// `bin/<triple>/<name>`). Lets callers detect (and gate) bundled drivers
220/// *before* copying or dispatching them — including binaries for other
221/// platforms, which are still bundled executable content.
222///
223/// # Errors
224/// Propagates read failures.
225pub fn bundled_drivers(payload_root: &Path) -> Result<Vec<String>, Error> {
226    let src = payload_root.join("drivers");
227    if !src.is_dir() {
228        return Ok(Vec::new());
229    }
230    let mut names = Vec::new();
231    for entry in std::fs::read_dir(&src).map_err(Error::io(format!("read {}", src.display())))? {
232        let entry = entry.map_err(Error::io("read driver entry"))?;
233        let file_type = entry
234            .file_type()
235            .map_err(Error::io(format!("stat {}", entry.path().display())))?;
236        let file_name = entry.file_name();
237        let Some(name) = file_name.to_str() else {
238            continue;
239        };
240        // `file_type` never follows symlinks — a linked `bin/` dir can't
241        // smuggle outside paths into the listing (mirrors registration).
242        if file_type.is_file() {
243            names.push(name.to_string());
244        } else if name == "bin" && file_type.is_dir() {
245            // `drivers/bin/<triple>/<name>` — compiled driver payloads.
246            names.extend(platform_bin_names(&entry.path())?);
247        }
248    }
249    names.sort();
250    Ok(names)
251}

253/// `bin/<triple>/<name>` labels for every regular file one level under
254/// each `drivers/bin/<triple>/` dir — all platforms, not just this host's.
255fn platform_bin_names(bin: &Path) -> Result<Vec<String>, Error> {
256    let mut names = Vec::new();
257    let Ok(triples) = std::fs::read_dir(bin) else {
258        return Ok(names);
259    };
260    for triple in triples {
261        let triple = triple.map_err(Error::io("read driver bin entry"))?;
262        if !triple
263            .file_type()
264            .map_err(Error::io(format!("stat {}", triple.path().display())))?
265            .is_dir()
266        {
267            continue;
268        }
269        let Some(triple_name) = triple.file_name().to_str().map(str::to_string) else {
270            continue;
271        };
272        for file in std::fs::read_dir(triple.path())
273            .map_err(Error::io(format!("read {}", triple.path().display())))?
274        {
275            let file = file.map_err(Error::io("read driver bin file"))?;
276            let is_file = file
277                .file_type()
278                .map_err(Error::io(format!("stat {}", file.path().display())))?
279                .is_file();
280            if is_file {
281                if let Some(name) = file.file_name().to_str() {
282                    names.push(format!("bin/{triple_name}/{name}"));
283                }
284            }
285        }
286    }
287    Ok(names)
288}

290/// Copy any bundled drivers a payload ships into `<tamias-home>/drivers/`
291/// so tamias discovers them: top-level `drivers/<name>` files as-is, plus
292/// the `drivers/bin/<target-triple>/` dir matching this host (see
293/// [`target_triple`]). Platform bins for other triples are skipped; a
294/// payload that ships platform dirs but none for this host is an error —
295/// the package intended a runnable compiled driver and can't provide one.
296/// Returns action lines.
297///
298/// # Errors
299/// Propagates read/copy failures, and the no-matching-platform error.
300pub fn register_bundled_drivers(payload_root: &Path) -> Result<Vec<String>, Error> {
301    let src = payload_root.join("drivers");
302    if !src.is_dir() {
303        return Ok(Vec::new());
304    }
305    let dest = drivers_dir();
306    std::fs::create_dir_all(&dest).map_err(Error::io(format!("create {}", dest.display())))?;
307    let mut actions = Vec::new();
308    let mut bin = None;
309    for entry in std::fs::read_dir(&src).map_err(Error::io(format!("read {}", src.display())))? {
310        let entry = entry.map_err(Error::io("read driver entry"))?;
311        // `file_type` never follows symlinks — a payload can't install a
312        // linked "driver" that resolves outside the payload.
313        let file_type = entry
314            .file_type()
315            .map_err(Error::io(format!("stat {}", entry.path().display())))?;
316        if file_type.is_dir() && entry.file_name() == "bin" {
317            bin = Some(entry.path());
318            continue;
319        }
320        if !file_type.is_file() {
321            continue;
322        }
323        let path = entry.path();
324        let name = path.file_name().unwrap_or_default();
325        let to = dest.join(name);
326        std::fs::copy(&path, &to).map_err(Error::io(format!("copy {}", path.display())))?;
327        set_executable(&to);
328        actions.push(format!("driver: {}", to.display()));
329    }
330    if let Some(bin) = bin {
331        actions.extend(register_platform_bins(&bin, &dest)?);
332    }
333    Ok(actions)
334}

336/// The `drivers/bin/<target-triple>/` payload: copy the dir matching this
337/// host into `dest` (flat, executable). Errors when platform dirs exist
338/// but none match — listing what the package does ship.
339fn register_platform_bins(bin: &Path, dest: &Path) -> Result<Vec<String>, Error> {
340    let mut triples = Vec::new();
341    for entry in std::fs::read_dir(bin).map_err(Error::io(format!("read {}", bin.display())))? {
342        let entry = entry.map_err(Error::io("read driver bin entry"))?;
343        if entry
344            .file_type()
345            .map_err(Error::io(format!("stat {}", entry.path().display())))?
346            .is_dir()
347        {
348            if let Some(name) = entry.file_name().to_str() {
349                triples.push(name.to_string());
350            }
351        }
352    }
353    triples.sort();
354    if triples.is_empty() {
355        return Ok(Vec::new());
356    }
357    let current = target_triple();
358    if !triples.iter().any(|t| t == current) {
359        return Err(Error::Failed(format!(
360            "package ships driver binaries for {} but not this platform ({current})",
361            triples.join(", ")
362        )));
363    }
364    let mut actions = Vec::new();
365    let platform = bin.join(current);
366    for entry in
367        std::fs::read_dir(&platform).map_err(Error::io(format!("read {}", platform.display())))?
368    {
369        let entry = entry.map_err(Error::io("read driver bin file"))?;
370        if !entry
371            .file_type()
372            .map_err(Error::io(format!("stat {}", entry.path().display())))?
373            .is_file()
374        {
375            continue;
376        }
377        let path = entry.path();
378        let name = path.file_name().unwrap_or_default();
379        let to = dest.join(name);
380        std::fs::copy(&path, &to).map_err(Error::io(format!("copy {}", path.display())))?;
381        set_executable(&to);
382        actions.push(format!("driver: {}", to.display()));
383    }
384    Ok(actions)
385}

387/// Best-effort exec bit on unix — mirrors the previous inline block.
388fn set_executable(path: &Path) {
389    #[cfg(unix)]
390    {
391        use std::os::unix::fs::PermissionsExt;
392        let _ = std::fs::set_permissions(path, std::fs::Permissions::from_mode(0o755));
393    }
394    #[cfg(not(unix))]
395    {
396        let _ = path;
397    }
398}

572    #[test]
573    fn platform_bins_select_current_triple() {
574        let _guard = ENV.lock().unwrap();
575        let tmp = tempfile::tempdir().unwrap();
576        std::env::set_var("TAMIAS_HOME", tmp.path().join("home"));
577        let payload = tmp.path().join("payload");
578        let current = target_triple();
579        // a triple that can never be this host's (rustc has no such target)
580        let foreign = "not-a-real-target-triple";
581        assert_ne!(current, foreign);
582        let cur = payload.join("drivers/bin").join(current);
583        let oth = payload.join("drivers/bin").join(foreign);
584        std::fs::create_dir_all(&cur).unwrap();
585        std::fs::create_dir_all(&oth).unwrap();
586        std::fs::write(cur.join("tamias-driver-cur"), "#!/bin/sh\nexit 0\n").unwrap();
587        std::fs::write(oth.join("tamias-driver-oth"), "#!/bin/sh\nexit 0\n").unwrap();
588        // interpreted driver alongside the bins still registers
589        std::fs::create_dir_all(payload.join("drivers")).unwrap();
590        std::fs::write(payload.join("drivers/tamias-driver-script"), "#!/bin/sh\n").unwrap();
591        let actions = register_bundled_drivers(&payload).unwrap();
592        assert_eq!(actions.len(), 2);
593        assert!(drivers_dir().join("tamias-driver-cur").is_file());
594        assert!(drivers_dir().join("tamias-driver-script").is_file());
595        // the foreign-platform binary is never copied
596        assert!(!drivers_dir().join("tamias-driver-oth").exists());
597        // exec bit set on the registered bin
598        #[cfg(unix)]
599        {
600            use std::os::unix::fs::PermissionsExt;
601            let mode = std::fs::metadata(drivers_dir().join("tamias-driver-cur"))
602                .unwrap()
603                .permissions()
604                .mode();
605            assert_eq!(mode & 0o111, 0o111);
606        }
607        // consent detection sees both forms, labeled by platform
608        let shipped = bundled_drivers(&payload).unwrap();
609        assert!(shipped.contains(&"tamias-driver-script".to_string()));
610        assert!(shipped.contains(&format!("bin/{current}/tamias-driver-cur")));
611        assert!(shipped.contains(&format!("bin/{foreign}/tamias-driver-oth")));
612        std::env::remove_var("TAMIAS_HOME");
613    }
