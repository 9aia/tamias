1//! views — the driver's manifest: every ViewSpec plus the query
2//! resolvers that feed them.
3//!
4//! Read path: all data comes from [`crate::data::Snapshot`] — one
5//! scan of the resolved tamias roots per wire op, through
6//! `tamias-store-local`/`tamias-cache-local`/`tamias-registry-local`/
7//! `tamias-driver` exactly like the `tamias` CLI verbs. Nothing here
8//! proxies the `tamias` binary or `tamias serve`: the reads are
9//! in-process dir walks over `.tamias-store.json` records, and
10//! keeping the driver a pure reader means it can never mutate the
11//! store, cache, or registry.
12//!
13//! Verb → view mapping:
14//!
15//! | `tamias` surface                  | pavo view                |
16//! |----------------------------------|---------------------------|
17//! | `tamias list` / `tamias info`     | `tamias:packages` table   |
18//! | `tamias info <name>`              | `tamias:package` (`--param name=`) |
19//! | content types + `tamias-driver-*` | `tamias:types` table      |
20//! | `tamias verify`                   | `tamias:verify` table     |
21//! | `tamias cache clean` stats        | `tamias:cache` kv         |
22//! | kind → driver dispatch           | `tamias:graph` graph      |
23//! | roots + counts                    | `tamias:status` kv        |
24

282/// Every query name the driver answers — the `query` op rejects
283/// anything outside this set with `ok:false`.
284pub fn known_query(name: &str) -> bool {
285    matches!(
286        name,
287        "summary" | "packages" | "package" | "types" | "verify" | "cache" | "graph"
288    )
289}

291/// `{"op":"query","query":name,"params":{…}}` → the data payload.
292/// Unknown names return an `{"error": …}` object (defensive — the
293/// wire layer rejects them before this runs).
294pub fn query_data(name: &str, params: &Map<String, Value>, snap: &Snapshot) -> Value {
295    if !snap.ok() {
296        return snap.error_payload();
297    }
298    match name {
299        "summary" => summary(snap),
300        "packages" => packages_rows(snap),
301        "package" => package_detail(snap, params),
302        "types" => types_rows(snap),
303        "verify" => verify_rows(snap, params),
304        "cache" => cache_payload(snap, params),
305        "graph" => graph_payload(snap),
306        other => json!({"error": format!("unknown query '{other}'")}),
307    }
308}

312/// Unix seconds → `YYYY-MM-DD HH:MM:SS` UTC — no chrono dep.
313fn fmt_ts(ts: u64) -> String {
314    let days = (ts / 86_400) as i64;
315    let secs = ts % 86_400;
316    let (h, m, s) = (secs / 3600, (secs % 3600) / 60, secs % 60);
317    // civil-from-days (Howard Hinnant's algorithm)
318    let z = days + 719_468;
319    let era = if z >= 0 { z } else { z - 146_096 } / 146_097;
320    let doe = z - era * 146_097;
321    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
322    let mut y = yoe + era * 400;
323    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
324    let mp = (5 * doy + 2) / 153;
325    let d = doy - (153 * mp + 2) / 5 + 1;
326    let mo = if mp < 10 { mp + 3 } else { mp - 9 };
327    if mo <= 2 {
328        y += 1;
329    }
330    format!("{y:04}-{mo:02}-{d:02} {h:02}:{m:02}:{s:02}")
331}

333/// Short digest — `sha256:abcdef…` → `abcdef123456`.
334fn short_digest(digest: &str) -> String {
335    let s = digest.strip_prefix("sha256:").unwrap_or(digest);
336    s.chars().take(12).collect()
337}

354/// `summary` query — the kv object: roots + counts + warnings.
355fn summary(snap: &Snapshot) -> Value {
356    let mut out = Map::new();
357    out.insert("home".into(), json!(snap.roots.home.display().to_string()));
358    out.insert(
359        "store".into(),
360        json!(snap.roots.store.display().to_string()),
361    );
362    out.insert(
363        "cache".into(),
364        json!(snap.roots.cache.display().to_string()),
365    );
366    out.insert(
367        "registry".into(),
368        json!(snap.roots.registry.display().to_string()),
369    );
370    out.insert(
371        "drivers_dir".into(),
372        json!(snap.roots.drivers.display().to_string()),
373    );
374    out.insert("roots_source".into(), json!(snap.roots.source));
375    out.insert("packages".into(), json!(snap.packages.len()));
376    let mut types: Vec<&str> = snap.packages.iter().map(kind_of).collect();
377    types.sort_unstable();
378    types.dedup();
379    out.insert("content_types".into(), json!(types));
380    out.insert("drivers".into(), json!(snap.drivers.len()));
381    out.insert("cache_entries".into(), json!(snap.cache_entries));
382    out.insert("registry_entries".into(), json!(snap.registry.len()));
383    out.insert("warnings".into(), json!(snap.warnings));
384    Value::Object(out)
385}

387/// `packages` query — one row per store entry (`tamias list` +
388/// provenance columns).
389fn packages_rows(snap: &Snapshot) -> Value {
390    Value::Array(
391        snap.packages
392            .iter()
393            .map(|e| {
394                let r = &e.record;
395                json!({
396                    "label": e.label(),
397                    "name": r.name,
398                    "scope": r.scope,
399                    "type": kind_of(e),
400                    "version": r.version,
401                    "source": r.source_ref,
402                    "resolved": r.resolved,
403                    "installed": fmt_ts(r.installed_at),
404                    "digest": short_digest(&r.digest),
405                })
406            })
407            .collect(),
408    )
409}

411/// `package` query — one entry's manifest + provenance as a kv
412/// object (`tamias info`). Param `name` — exact label, `@scope/name`,
413/// or bare name (ambiguity errors surface as `{"error": …}`).
414fn package_detail(snap: &Snapshot, params: &Map<String, Value>) -> Value {
415    let Some(name) = param(params, "name") else {
416        return json!({"error": "missing required param 'name' (e.g. --param name=myco-config)"});
417    };
418    // `Store::find` handles exact labels, bare names, and the
419    // several-installs ambiguity error — same lookup `tamias info`.
420    let store = tamias_store_local::Store::new(&snap.roots.store);
421    let entry = match store.find(&name) {
422        Ok(e) => e,
423        Err(err) => return json!({"error": err.to_string()}),
424    };
425    let r = &entry.record;
426    let mut out = Map::new();
427    out.insert("label".into(), json!(entry.label()));
428    out.insert("name".into(), json!(r.name));
429    if let Some(scope) = &r.scope {
430        out.insert("scope".into(), json!(scope));
431    }
432    out.insert(
433        "version".into(),
434        json!(r.version.clone().unwrap_or_else(|| "-".into())),
435    );
436    out.insert("type".into(), json!(kind_of(&entry)));
437    out.insert("source_ref".into(), json!(r.source_ref));
438    out.insert("resolved".into(), json!(r.resolved));
439    out.insert("digest".into(), json!(r.digest));
440    out.insert("installed_at".into(), json!(fmt_ts(r.installed_at)));
441    out.insert("dir".into(), json!(entry.dir.display().to_string()));
442    match entry.size() {
443        Ok(size) => {
444            out.insert("size_bytes".into(), json!(size));
445        }
446        Err(e) => {
447            out.insert("size_error".into(), json!(e.to_string()));
448        }
449    }
450    // Payload manifest — description/include/deps when present.
451    match entry.manifest() {
452        Ok(m) => {
453            if let Some(d) = &m.description {
454                out.insert("description".into(), json!(d));
455            }
456            if !m.include.is_empty() {
457                out.insert("include".into(), json!(m.include));
458            }
459            if !m.deps.is_empty() {
460                out.insert("deps".into(), json!(m.deps));
461            }
462        }
463        Err(e) => {
464            out.insert("manifest_error".into(), json!(e.to_string()));
465        }
466    }
467    Value::Object(out)
468}

470/// The types a driver executable claims — filename kind plus the
471/// `capabilities` `types` array (same probe `tamias-driver`'s
472/// `driver_handles` runs). Probe failures degrade to filename-only.
473fn driver_claims(path: &std::path::Path, filename_kind: &str) -> Vec<String> {
474    let mut kinds = vec![filename_kind.to_string()];
475    let Ok(out) = std::process::Command::new(path)
476        .arg("capabilities")
477        .output()
478    else {
479        return kinds;
480    };
481    if !out.status.success() {
482        return kinds;
483    }
484    let Ok(value) = serde_json::from_slice::<Value>(&out.stdout) else {
485        return kinds;
486    };
487    if let Some(types) = value.get("types").and_then(Value::as_array) {
488        for t in types.iter().filter_map(Value::as_str) {
489            if !kinds.iter().any(|k| k == t) {
490                kinds.push(t.to_string());
491            }
492        }
493    }
494    kinds
495}

497/// `types` query — content types in play: every kind installed
498/// packages declare ∪ every kind a discovered driver claims.
499fn types_rows(snap: &Snapshot) -> Value {
500    // type → (package count, driver paths claiming it)
501    let mut table: std::collections::BTreeMap<String, (usize, Vec<String>)> =
502        std::collections::BTreeMap::new();
503    for e in &snap.packages {
504        table.entry(kind_of(e).to_string()).or_default().0 += 1;
505    }
506    for (kind, path) in &snap.drivers {
507        for claimed in driver_claims(path, kind) {
508            table
509                .entry(claimed)
510                .or_default()
511                .1
512                .push(path.display().to_string());
513        }
514    }
515    // Drivers claiming nothing still show under their filename kind.
516    Value::Array(
517        table
518            .into_iter()
519            .map(|(t, (packages, drivers))| {
520                json!({
521                    "type": t,
522                    "packages": packages,
523                    "drivers": drivers.len(),
524                    "driver_paths": drivers,
525                })
526            })
527            .collect(),
528    )
529}

531/// `verify` query — per-package digest status.
532///
533/// `mode=quick` (default): the record carries a `sha256:` digest and
534/// the payload dir still exists — O(1) per entry, what a status
535/// surface can afford. `mode=full`: [`Store::verify`] recomputes
536/// `dir_digest` over every payload file — identical to
537/// `tamias verify <name>`; expensive on large stores, opt-in via
538/// `--param mode=full`. `name=` limits the check to one package.
539fn verify_rows(snap: &Snapshot, params: &Map<String, Value>) -> Value {
540    let full = param(params, "mode").is_some_and(|m| m == "full" || m == "true");
541    let only = param(params, "name");
542    let store = tamias_store_local::Store::new(&snap.roots.store);
543    let mut rows = Vec::new();
544    for e in &snap.packages {
545        if let Some(n) = &only {
546            if e.label() != *n && e.record.name != *n {
547                continue;
548            }
549        }
550        let recorded = e.record.digest.clone();
551        let well_formed = recorded.starts_with("sha256:") && recorded.len() > 7;
552        let dir_exists = e.dir.is_dir();
553        let (status, detail) = if full {
554            match store.verify(&e.label()) {
555                Ok(report) => {
556                    if report.ok {
557                        (
558                            "ok".to_string(),
559                            format!("recomputed {}", short_digest(&report.computed)),
560                        )
561                    } else {
562                        (
563                            "mismatch".to_string(),
564                            format!(
565                                "recorded {} computed {}",
566                                short_digest(&report.recorded),
567                                short_digest(&report.computed)
568                            ),
569                        )
570                    }
571                }
572                Err(err) => ("error".to_string(), err.to_string()),
573            }
574        } else if !well_formed {
575            (
576                "no-digest".to_string(),
577                "record carries no sha256 digest".to_string(),
578            )
579        } else if !dir_exists {
580            (
581                "missing-dir".to_string(),
582                format!("payload dir gone: {}", e.dir.display()),
583            )
584        } else {
585            (
586                "recorded".to_string(),
587                "digest present; payload dir exists".to_string(),
588            )
589        };
590        rows.push(json!({
591            "label": e.label(),
592            "name": e.record.name,
593            "recorded": short_digest(&recorded),
594            "status": status,
595            "detail": detail,
596        }));
597    }
598    if let Some(n) = &only {
599        if rows.is_empty() {
600            rows.push(json!({
601                "label": n,
602                "name": n,
603                "recorded": "",
604                "status": "not-installed",
605                "detail": "no store entry matches",
606            }));
607        }
608    }
609    Value::Array(rows)
610}

612/// `cache` query — fetch-cache kv object. Bytes only when the
613/// snapshot computed them (`size`/`detail` param on the op).
614fn cache_payload(snap: &Snapshot, _params: &Map<String, Value>) -> Value {
615    let mut out = Map::new();
616    out.insert("root".into(), json!(snap.roots.cache.display().to_string()));
617    out.insert("entries".into(), json!(snap.cache_entries));
618    match snap.cache_bytes {
619        Some(bytes) => {
620            out.insert("bytes".into(), json!(bytes));
621        }
622        None => {
623            out.insert(
624                "bytes".into(),
625                json!("not computed — pass size=1 to total payload bytes"),
626            );
627        }
628    }
629    out.insert(
630        "note".into(),
631        json!("content-addressed fetch cache — `tamias cache clean [--all]` evicts; entries are immutable blobs"),
632    );
633    Value::Object(out)
634}

636/// `graph` query — `{"nodes":[…],"edges":[…]}`: installed packages
637/// and discovered drivers, with an edge wherever a driver claims the
638/// package's content type (the dispatch `tamias install` performs).
639fn graph_payload(snap: &Snapshot) -> Value {
640    let mut nodes: Vec<Value> = Vec::new();
641    let mut edges: Vec<Value> = Vec::new();
642    // Driver nodes — one per discovered executable; claims = filename
643    // kind + capabilities.types (same probe as the types view).
644    let mut driver_claims_map: Vec<(String, Vec<String>)> = Vec::new();
645    for (kind, path) in &snap.drivers {
646        let claims = driver_claims(path, kind);
647        driver_claims_map.push((kind.clone(), claims.clone()));
648        nodes.push(json!({
649            "id": format!("drv:{kind}"),
650            "label": format!("tamias-driver-{kind}"),
651            "kind": "driver",
652            "props": {"path": path.display().to_string(), "claims": claims},
653        }));
654    }
655    for e in &snap.packages {
656        let pkg_kind = kind_of(e).to_string();
657        nodes.push(json!({
658            "id": format!("pkg:{}", e.label()),
659            "label": e.label(),
660            "kind": "package",
661            "props": {"type": pkg_kind, "source_ref": e.record.source_ref},
662        }));
663        for (dkind, claims) in &driver_claims_map {
664            if claims.contains(&pkg_kind) {
665                edges.push(json!({
666                    "from": format!("pkg:{}", e.label()),
667                    "to": format!("drv:{dkind}"),
668                    "label": pkg_kind,
669                }));
670            }
671        }
672    }
673    json!({"nodes": nodes, "edges": edges})
674}

676// ---------------------------------------------------------------- view op
677
678/// `{"op":"view","view":id,"params":{…}}` → [`pavo_protocol::wire::ViewResult`].
679pub fn view_result(
680    view_id: &str,
681    params: &Map<String, Value>,
682) -> Result<pavo_protocol::wire::ViewResult, String> {
683    let spec = specs()
684        .into_iter()
685        .find(|v| v.id == view_id)
686        .ok_or_else(|| format!("unknown view: {view_id}"))?;
687
688    let snap = crate::data::load(params);
689    let mut data = Map::new();
690    for b in &spec.queries {
691        data.insert(b.name.clone(), query_data(&b.name, params, &snap));
692    }
693
694    Ok(pavo_protocol::wire::ViewResult { view: spec, data })
695}
