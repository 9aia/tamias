2//! tamias-cache-local — the content-addressed fetch cache:
3//! `<root>/<key>/` payloads (`git-<sha>`, `sha256-<hex>`). Install is
4//! fetch→unpack, so a warm cache makes repeat installs and air-gapped
5//! nodes free. `remove` never touches it; eviction is by age or full.
6
7use std::fs;
8use std::path::{Path, PathBuf};
9use std::time::{Duration, SystemTime};
10
11use tamias_core::Error;
12
13/// Cache keys are `[A-Za-z0-9._-]+`.
14fn valid_key(key: &str) -> bool {
15    !key.is_empty()
16        && key
17            .bytes()
18            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-'))
19        && key != "."
20        && key != ".."
21}
22
23/// The content-addressed fetch cache — one per root.
24pub struct FetchCache {
25    root: PathBuf,
26}
27
28impl FetchCache {
29    /// A cache under `root` (created lazily on first commit).
30    #[must_use]
31    pub fn new(root: impl Into<PathBuf>) -> Self {
32        Self { root: root.into() }
33    }
34
35    /// The root dir.
36    #[must_use]
37    pub fn root(&self) -> &Path {
38        &self.root
39    }
40
41    /// Path for a key — caller checks [`FetchCache::has`].
42    #[must_use]
43    pub fn entry(&self, key: &str) -> PathBuf {
44        self.root.join(key)
45    }
46
47    /// Whether a populated entry exists for `key`.
48    #[must_use]
49    pub fn has(&self, key: &str) -> bool {
50        valid_key(key) && self.entry(key).is_dir()
51    }
52
53    /// A scratch dir for an in-flight fetch.
54    ///
55    /// # Errors
56    /// Propagates I/O failures creating the scratch dir.
57    pub fn scratch(&self) -> Result<PathBuf, Error> {
58        let ctx = format!("create {}", self.root.display());
59        fs::create_dir_all(&self.root).map_err(Error::io(ctx))?;
60        let scratch = self.root.join(format!(".scratch-{}", std::process::id()));
61        if scratch.exists() {
62            let clear = format!("clear {}", scratch.display());
63            fs::remove_dir_all(&scratch).map_err(Error::io(clear))?;
64        }
65        fs::create_dir_all(&scratch).map_err(Error::io(format!("create {}", scratch.display())))?;
66        Ok(scratch)
67    }
68
69    /// Fold a filled scratch dir into `entry(key)` — an existing entry wins.
70    ///
71    /// # Errors
72    /// Fails on an invalid key or I/O error.
73    pub fn commit(&self, scratch: &Path, key: &str) -> Result<PathBuf, Error> {
74        if !valid_key(key) {
75            return Err(Error::Failed(format!("invalid cache key '{key}'")));
76        }
77        let dest = self.entry(key);
78        if dest.exists() {
79            let clear = format!("clear {}", scratch.display());
80            fs::remove_dir_all(scratch).map_err(Error::io(clear))?;
81            return Ok(dest);
82        }
83        let ctx = format!("commit {} → {}", scratch.display(), dest.display());
84        fs::rename(scratch, &dest).map_err(Error::io(ctx))?;
85        Ok(dest)
86    }
87
88    /// Abandon a scratch dir after a failed fetch.
89    pub fn discard(&self, scratch: &Path) {
90        if scratch.exists() {
91            let _ = fs::remove_dir_all(scratch);
92        }
93    }
94
95    /// Entry dirs (dotfiles excluded — scratches aren't entries).
96    ///
97    /// # Errors
98    /// Propagates read failures.
99    pub fn entries(&self) -> Result<Vec<PathBuf>, Error> {
100        let mut out = Vec::new();
101        if self.root.is_dir() {
102            let ctx = format!("read {}", self.root.display());
103            let read = fs::read_dir(&self.root).map_err(Error::io(ctx))?;
104            for entry in read {
105                let path = entry.map_err(Error::io("read cache entry"))?.path();
106                let hidden = path
107                    .file_name()
108                    .and_then(|n| n.to_str())
109                    .is_some_and(|n| n.starts_with('.'));
110                if path.is_dir() && !hidden {
111                    out.push(path);
112                }
113            }
114        }
115        out.sort();
116        Ok(out)
117    }
118
119    /// Evict entries whose mtime is older than `max_age`.
120    ///
121    /// # Errors
122    /// Propagates read/remove failures.
123    pub fn clean(&self, max_age: Duration) -> Result<Vec<PathBuf>, Error> {
124        let now = SystemTime::now();
125        let mut evicted = Vec::new();
126        for path in self.entries()? {
127            let stale = fs::metadata(&path)
128                .and_then(|m| m.modified())
129                .is_ok_and(|mtime| now.duration_since(mtime).is_ok_and(|age| age >= max_age));
130            if stale {
131                let ctx = format!("remove {}", path.display());
132                fs::remove_dir_all(&path).map_err(Error::io(ctx))?;
133                evicted.push(path);
134            }
135        }
136        Ok(evicted)
137    }
138
139    /// Full eviction — `cache clean --all`.
140    ///
141    /// # Errors
142    /// Propagates read/remove failures.
143    pub fn clean_all(&self) -> Result<Vec<PathBuf>, Error> {
144        self.clean(Duration::ZERO)
145    }
146}

155    #[test]
156    fn scratch_commit_has_clean() {
157        let root = tempfile::tempdir().unwrap();
158        let cache = FetchCache::new(root.path());
159        assert!(!cache.has("sha256-x"));
160        let scratch = cache.scratch().unwrap();
161        std::fs::write(scratch.join("f"), "x").unwrap();
162        cache.commit(&scratch, "sha256-x").unwrap();
163        assert!(cache.has("sha256-x"));
164        assert_eq!(cache.entries().unwrap().len(), 1);
165        assert_eq!(cache.clean_all().unwrap().len(), 1);
166        assert!(!cache.has("sha256-x"));
167    }

170    fn invalid_key_and_discard() {
171        let root = tempfile::tempdir().unwrap();
172        let cache = FetchCache::new(root.path().join("fetch"));
173        assert!(cache
174            .commit(std::path::Path::new("/nope"), "bad/key")
175            .is_err());
176        let scratch = cache.scratch().unwrap();
177        cache.discard(&scratch);
178        assert!(!scratch.exists());
179        cache.discard(std::path::Path::new("/nonexistent"));

230    #[test]
231    fn io_error_arms() {
232        use std::os::unix::fs::PermissionsExt;
233        let base = tempfile::tempdir().unwrap();
234        // root's parent read-only → scratch() create fails
235        let ro = base.path().join("ro");
236        std::fs::create_dir_all(&ro).unwrap();
237        std::fs::set_permissions(&ro, std::fs::Permissions::from_mode(0o555)).unwrap();
238        let cache = FetchCache::new(ro.join("cache"));
239        assert!(cache.scratch().is_err());
240        // unreadable root → entries()/clean() read fail; commit rename fails
241        let r2 = base.path().join("r2");
242        std::fs::create_dir_all(r2.join("k1")).unwrap();
243        let c2 = FetchCache::new(&r2);
244        assert_eq!(c2.entries().unwrap().len(), 1);
245        std::fs::set_permissions(&r2, std::fs::Permissions::from_mode(0o000)).unwrap();
246        assert!(c2.entries().is_err());
247        assert!(c2.clean(Duration::from_secs(0)).is_err());
248        assert!(c2.commit(&c2.entry("nope"), "k2").is_err());
249        std::fs::set_permissions(&r2, std::fs::Permissions::from_mode(0o755)).unwrap();
250        std::fs::set_permissions(&ro, std::fs::Permissions::from_mode(0o755)).unwrap();
251    }
