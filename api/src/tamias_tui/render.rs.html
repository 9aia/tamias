19/// Accent — a bright saturated blue that stays legible on dark
20/// terminals (the old `Indexed(69)` navy read as near-background);
21/// also the active tab's background and the `›` cursor marker.
22const ACCENT: Color = Color::Indexed(75);
23/// Secondary text — light gray: `Indexed(245)` read as near-bg on
24/// dark themes (the "almost black" secondary copy); `Indexed(250)`
25/// stays clearly a step dimmer than [`TEXT`] without sinking.
26const MUTED: Color = Color::Indexed(250);
27/// Primary text — the explicit fg for spans that carry no style of
28/// their own; terminals' default fg can be near-black on dark
29/// themes, which is what made table cells and body copy vanish.
30const TEXT: Color = Color::White;
31const WARN: Color = Color::Indexed(214);

33/// Chrome budget — `(tabs, status)`: tabs cost a row at height ≥3,
34/// the status bar only at height ≥5.
35fn chrome(height: u16) -> (bool, bool) {
36    (height >= 3, height >= 5)
37}

39/// One full frame. Also records the tab/pane geometry on `m` so the
40/// next mouse event can hit-test against where rows landed.
41pub fn draw(f: &mut Frame, m: &mut Model) {
42    let area = f.area();
43    // `L` docks the op log as a right-hand sidebar — the pane chrome
44    // lays out inside the shrunken main rect, so `tabs_area`/
45    // `board_area`/`home_area` and the status bar all describe what
46    // is actually on screen (the status bar shrinks with the pane,
47    // it doesn't span under the sidebar). The dock needs ~96 cols:
48    // below that the sidebar would starve the pane (the Installed
49    // table wants ~66 for its columns) and the centered overlay
50    // takes over. The `saturating_sub(70)` cap keeps the main
51    // column ≥70 even right at the boundary.
52    let split = if m.mode == Mode::Log && area.width >= 96 {
53        let side_w = (area.width / 4)
54            .clamp(24, 48)
55            .min(area.width.saturating_sub(70));
56        let c = Layout::horizontal([Constraint::Min(0), Constraint::Length(side_w)]).split(area);
57        Some((c[0], c[1]))
58    } else {
59        None
60    };
61    let main = split.map_or(area, |(main, _)| main);
62    let (show_tabs, show_status) = chrome(main.height);
63    let mut constraints = Vec::with_capacity(3);
64    if show_tabs {
65        constraints.push(Constraint::Length(1));
66    }
67    constraints.push(Constraint::Min(0));
68    if show_status {
69        constraints.push(Constraint::Length(1));
70    }
71    let chunks = Layout::vertical(constraints).split(main);
72
73    let mut next = 0;
74    if show_tabs {
75        draw_tabs(f, m, chunks[next]);
76        m.tabs_area = Some(chunks[next]);
77        next += 1;
78    } else {
79        m.tabs_area = None;
80    }
81    draw_pane(f, m, chunks[next]);
82    next += 1;
83    if show_status {
84        draw_status(f, m, chunks[next]);
85    }
86    // Help scrolls, so it needs `&mut Model` (clamps `help_scroll`,
87    // records the view height/max) — it can't ride the `&m.mode`
88    // match the read-only overlays share.
89    if m.mode == Mode::Help {
90        draw_help(f, m, area);
91    } else {
92        match &m.mode {
93            Mode::Filter | Mode::Search | Mode::Command | Mode::Prompt(_) => draw_input(f, m, area),
94            _ => {}
95        }
96    }
97    // The log view needs `&mut Model` (it clamps `log_scroll` and
98    // reports `log_view_h`/`log_area`), so it can't ride the
99    // `&m.mode` match.
100    if let Some((_, side)) = split {
101        draw_log_view(f, m, side);
102    } else if m.mode == Mode::Log {
103        draw_log_overlay(f, m, area);
104    }
105    if let Some(modal) = &m.modal {
106        draw_modal(f, modal, area);
107    }
108}

110/// Tab captions — full names when they fit, initials when they
111/// don't (the `1`-`6`/`h`/`l` binds still work; PROJECT/PUBLISH get
112/// two-char codes since both start with P).
113fn tab_labels(width: usize) -> Vec<String> {
114    let full: Vec<String> = Pane::ALL
115        .iter()
116        .map(|p| format!(" {} ", p.label()))
117        .collect();
118    let need: usize = full.iter().map(|s| s.chars().count()).sum::<usize>() + full.len() + 1;
119    if need <= width {
120        full
121    } else {
122        // Initials fallback — PROJECT and PUBLISH collide on 'P',
123        // so those two widen to two chars ("PR"/"PU"); the digits
124        // and h/l binds don't depend on the letters anyway.
125        Pane::ALL
126            .iter()
127            .map(|p| match p {
128                Pane::Manifest => "PR".to_string(),
129                Pane::Author => "PU".to_string(),
130                _ => p.label()[..1].to_string(),
131            })
132            .collect()
133    }
134}

136fn draw_tabs(f: &mut Frame, m: &Model, area: Rect) {
137    let mut spans = vec![Span::raw(" ")];
138    for (p, label) in Pane::ALL.iter().zip(tab_labels(area.width as usize)) {
139        let style = if *p == m.pane {
140            Style::default().fg(Color::White).bg(ACCENT)
141        } else {
142            Style::default().fg(MUTED)
143        };
144        spans.push(Span::styled(label, style));
145        spans.push(Span::raw(" "));
146    }
147    f.render_widget(Paragraph::new(Line::from(spans)), area);
148}

150/// Click hit-test for the tab row: which tab does column `x` land
151/// on in a `width`-wide bar? Mirrors the span layout `draw_tabs`
152/// emits — leading space + one separator per tab.
153pub fn tab_hit(width: u16, x: u16) -> Option<usize> {
154    let mut col = 1u16; // leading space
155    for (i, label) in tab_labels(width as usize).iter().enumerate() {
156        let w = label.chars().count() as u16;
157        if (col..col + w).contains(&x) {
158            return Some(i);
159        }
160        col += w + 1; // the separator after each tab
161    }
162    None
163}

165fn human_size(bytes: u64) -> String {
166    const UNITS: [&str; 5] = ["B", "KiB", "MiB", "GiB", "TiB"];
167    let mut v = bytes as f64;
168    let mut u = 0;
169    while v >= 1024.0 && u + 1 < UNITS.len() {
170        v /= 1024.0;
171        u += 1;
172    }
173    if u == 0 {
174        format!("{bytes} B")
175    } else {
176        format!("{v:.1} {}", UNITS[u])
177    }
178}

185/// Unix seconds now — [`ago`]'s clock.
186fn unix_now() -> u64 {
187    SystemTime::now()
188        .duration_since(UNIX_EPOCH)
189        .map(|d| d.as_secs())
190        .unwrap_or(0)
191}

193/// Relative-time label for an `installed_at` unix-seconds stamp —
194/// "just now", "5m ago", "3h ago", "2d ago", then months/years.
195/// `now` is a parameter so the buckets stay unit-testable; [`ago`]
196/// is the clock-reading wrapper the Home pane uses.
197fn ago_at(ts: u64, now: u64) -> String {
198    let secs = now.saturating_sub(ts);
199    match secs {
200        0..=59 => "just now".to_string(),
201        60..=3599 => format!("{}m ago", secs / 60),
202        3600..=86_399 => format!("{}h ago", secs / 3600),
203        86_400..=2_592_000 => format!("{}d ago", secs / 86_400), // ≤30d
204        2_592_001..=31_536_000 => format!("{}mo ago", secs / 2_592_000),
205        _ => format!("{}y ago", secs / 31_536_000),
206    }
207}
208
209/// [`ago_at`] against the wall clock.
210fn ago(ts: u64) -> String {
211    ago_at(ts, unix_now())
212}

214/// Trust-tier color — `signed`/`official` green, `bad` red,
215/// `pinned`/`unpinned` warn yellow, anything else muted.
216fn trust_style(tier: &str) -> Style {
217    match tier {
218        "official" | "signed" => Style::default().fg(Color::Green),
219        "bad" => Style::default().fg(Color::Red),
220        "pinned" | "unpinned" => Style::default().fg(WARN),
221        _ => Style::default().fg(MUTED),
222    }
223}

232/// The selection-marker cell the data tables prepend as a narrow
233/// first column — accent `›` on the cursor row, blank elsewhere —
234/// so the cursor reads as clearly as Home's recents and the
235/// versions-focus list (the `Indexed(238)` row bg alone was too
236/// subtle). Row clicks map by display row, so the extra column
237/// doesn't disturb hit-testing.
238fn sel_cell(pos: usize, sel: usize) -> Cell<'static> {
239    if pos == sel {
240        Cell::from(Line::from(Span::styled("›", Style::default().fg(ACCENT))))
241    } else {
242        Cell::from("")
243    }

246fn draw_pane(f: &mut Frame, m: &mut Model, area: Rect) {
247    match m.pane {
248        // The package page replaces whichever pane opened it — the
249        // model only sets `detail` on Installed/Registry and clears
250        // it on pane switches.
251        Pane::Installed | Pane::Registry if m.detail.is_some() => draw_detail(f, m, area),
252        Pane::Home => draw_home(f, m, area),
253        Pane::Installed => draw_installed(f, m, area),
254        Pane::Registry => draw_registry(f, m, area),
255        Pane::Services => draw_services(f, m, area),
256        Pane::Manifest => draw_manifest(f, m, area),
257        Pane::Author => draw_author(f, m, area),
258    }
259}

261/// The Home pane — the npm-home analog and the landing view:
262/// wordmark + version, a muted stats strip (installed count, registry
263/// target, store/cache roots), and the recently-installed list
264/// (`installed_at` desc, top 8 — the same
265/// [`Model::recent_installed`] ordering the `home_sel` cursor walks).
266/// Key hints live only in the status bar — the body deliberately
267/// carries none (the footer is the single hint surface).
268/// The highlighted row gets the versions-focus treatment — accent
269/// `›` marker + `Indexed(238)` row background — and `Enter` opens its
270/// package page via the Installed pane.
271///
272/// The recents block is pinned to the bottom of the inner area and
273/// rendered without wrapping, so each row is exactly one display
274/// line — that's what lets `Model::home_area` map a mouse click to a
275/// `recent_installed()` index. The wrapping header Paragraph takes
276/// whatever height is left above it.
277fn draw_home(f: &mut Frame, m: &mut Model, area: Rect) {
278    m.home_area = Rect::default();
279    let mut lines: Vec<Line> = Vec::new();
280    // Wordmark — accent dot + bold white name; the crate version
281    // rides beside it, the tagline on its own muted line.
282    lines.push(Line::from(vec![
283        Span::styled("● ", Style::default().fg(ACCENT)),
284        Span::styled(
285            "tamias",
286            Style::default()
287                .fg(Color::White)
288                .add_modifier(Modifier::BOLD),
289        ),
290        Span::styled(
291            format!("  v{}", env!("CARGO_PKG_VERSION")),
292            Style::default().fg(MUTED),
293        ),
294    ]));
295    lines.push(Line::from(Span::styled(
296        "package & content manager",
297        Style::default().fg(MUTED),
298    )));
299    lines.push(Line::from(""));
300    // Stats — one muted line that wraps naturally on narrow frames.
301    // Superseded store versions append the `:prune` nudge (a status
302    // *value*, not a keybind legend — the footer stays the only hint
303    // surface).
304    let prune = m.prune_removals();
305    let prune_note = if prune > 0 {
306        format!(
307            " · {prune} old version{} — :prune",
308            if prune == 1 { "" } else { "s" }
309        )
310    } else {
311        String::new()
312    };
313    lines.push(Line::from(Span::styled(
314        format!(
315            "{} installed{} · registry {} · store {} · cache {}",
316            m.entries.len(),
317            prune_note,
318            m.reg_target_label,
319            m.ctx.store_root.display(),
320            m.ctx.cache_root.display()
321        ),
322        Style::default().fg(MUTED),
323    )));
324    lines.push(Line::from(""));
325    // The recents block — section header plus one line per row (or
326    // the empty-state hint). Kept out of `lines` so it renders in its
327    // own fixed-height chunk below.
328    let mut tail: Vec<Line> = vec![Line::from(Span::styled(
329        "recently installed",
330        Style::default().fg(ACCENT),
331    ))];
332    if m.entries.is_empty() {
333        tail.push(Line::from(Span::styled(
334            "  nothing installed yet",
335            Style::default().fg(MUTED),
336        )));
337    } else {
338        for (pos, &i) in m.recent_installed().iter().enumerate() {
339            let row = &m.entries[i];
340            let rec = &row.entry.record;
341            let cursor = pos == m.home_sel;
342            let mut spans = vec![Span::styled(
343                if cursor { "› " } else { "  " },
344                if cursor {
345                    Style::default().fg(ACCENT)
346                } else {
347                    Style::default()
348                },
349            )];
350            spans.push(Span::raw(row.entry.label()));
351            if let Some(badge) = trust_badge(rec.trust.as_deref()) {
352                spans.push(badge);
353            }
354            if rec.installed_at > 0 {
355                spans.push(Span::styled(
356                    format!("  · {}", ago(rec.installed_at)),
357                    Style::default().fg(MUTED),
358                ));
359            }
360            let line = Line::from(spans);
361            tail.push(if cursor {
362                line.style(Style::default().bg(Color::Indexed(238)))
363            } else {
364                line
365            });
366        }
367    }
368    let block = Block::default().borders(Borders::ALL).title(" home ");
369    let inner = block.inner(area);
370    f.render_widget(block, area);
371    if inner.height == 0 || inner.width == 0 {
372        m.board_area = Rect::default();
373        m.viewport_top = 0;
374        return;
375    }
376    // Pin the tail to the bottom — `Min(0)` header above it. The tail
377    // Paragraph gets no `wrap`, so each `Line` is exactly one display
378    // row (long labels clip instead of wrapping onto the next row).
379    let tail_h = (tail.len() as u16).min(inner.height);
380    let chunks = Layout::vertical([Constraint::Min(0), Constraint::Length(tail_h)]).split(inner);
381    f.render_widget(
382        Paragraph::new(lines)
383            .style(Style::default().fg(TEXT))
384            .wrap(Wrap { trim: true }),
385        chunks[0],
386    );
387    f.render_widget(
388        Paragraph::new(tail).style(Style::default().fg(TEXT)),
389        chunks[1],
390    );
391    // Data rows start one line below the "recently installed" header;
392    // height is the rows actually rendered (clipped on short frames).
393    let shown = chunks[1]
394        .height
395        .saturating_sub(1)
396        .min(m.recent_installed().len() as u16);
397    if shown > 0 {
398        m.home_area = Rect {
399            x: chunks[1].x,
400            y: chunks[1].y.saturating_add(1),
401            width: chunks[1].width,
402            height: shown,
403        };
404    }
405    m.board_area = Rect::default();
406    m.viewport_top = 0;
407}

409/// The Services pane — every installed provider service with its
410/// unit state. Rows come from [`Job::ServicesRefresh`]; the table
411/// marks stale while a refresh is queued/running via the busy bar.
412fn draw_services(f: &mut Frame, m: &mut Model, area: Rect) {
413    if let Some(err) = &m.svc_err {
414        if m.svc_rows.is_empty() {
415            f.render_widget(
416                Paragraph::new(format!(
417                    "services: {err} — `:services` retries · `:doctor` diagnoses"
418                ))
419                .style(Style::default().fg(Color::Red)),
420                area,
421            );
422            m.board_area = Rect::default();
423            m.viewport_top = 0;
424            return;
425        }
426    }
427    if m.svc_rows.is_empty() {
428        let hint = if m.busy.is_some() {
429            "loading services…"
430        } else {
431            "no provider services — a package's `provider:` block declares them"
432        };
433        f.render_widget(Paragraph::new(hint).style(Style::default().fg(MUTED)), area);
434        m.board_area = Rect::default();
435        m.viewport_top = 0;
436        return;
437    }
438    let header = Row::new(vec!["", "PACKAGE", "SERVICE", "UNIT", "STATE", "EXEC"])
439        .style(Style::default().fg(ACCENT).add_modifier(Modifier::BOLD));
440    // The package cell prints once per consecutive same-package run —
441    // `svc_rows` arrives sorted by package at ingest, so groups are
442    // contiguous; continuation rows leave it blank rather than
443    // repeating the label N times. Rows are already deduped to one
444    // per unit; `svc_counts` marks how many store entries provide
445    // the unit (`×N`, muted). `svc_sel` still indexes the flat row
446    // list (package lookups read the row, not the cell).
447    let mut rows: Vec<Row> = Vec::with_capacity(m.svc_rows.len());
448    let mut prev_pkg: Option<&str> = None;
449    for (pos, r) in m.svc_rows.iter().enumerate() {
450        let pkg = if prev_pkg == Some(r.package.as_str()) {
451            String::new()
452        } else {
453            r.package.clone()
454        };
455        prev_pkg = Some(r.package.as_str());
456        let mut pkg_spans = vec![Span::raw(pkg)];
457        let providers = m.svc_counts.get(pos).copied().unwrap_or(1);
458        if providers > 1 {
459            pkg_spans.push(Span::styled(
460                format!(" ×{providers}"),
461                Style::default().fg(MUTED),
462            ));
463        }
464        rows.push(Row::new(vec![
465            sel_cell(pos, m.svc_sel),
466            Cell::from(Line::from(pkg_spans)),
467            Cell::from(r.service.clone()),
468            Cell::from(r.unit.clone()),
469            Cell::from(r.state.clone().unwrap_or_else(|| "-".into())),
470            Cell::from(r.exec.clone()),
471        ]));
472    }
473    let widths = [
474        Constraint::Length(2),
475        Constraint::Min(14),
476        Constraint::Length(12),
477        Constraint::Min(14),
478        Constraint::Length(10),
479        Constraint::Min(16),
480    ];
481    let title = if m.svc_stale || m.busy.is_some() {
482        " services — refreshing… "
483    } else {
484        " services "
485    };
486    let table = Table::new(rows, widths)
487        .header(header)
488        .block(Block::default().borders(Borders::ALL).title(title))
489        .style(Style::default().fg(TEXT))
490        .row_highlight_style(Style::default().bg(Color::Indexed(238)));
491    let mut state = ratatui::widgets::TableState::default().with_selected(Some(m.svc_sel));
492    f.render_stateful_widget(table, area, &mut state);
493    note_table_area(m, area, state.offset());
494}

496/// The Project pane — the project manifest's `deps:` as editable
497/// rows; every other field renders read-only above the table. Writes
498/// go through `Manifest::save` + `tamias.lock` sync on the worker.
499fn draw_manifest(f: &mut Frame, m: &mut Model, area: Rect) {
500    // Read-only header — path plus the manifest's scalar fields.
501    let mut info: Vec<Line> = Vec::new();
502    match (&m.project, &m.man_err) {
503        (Some(manifest), _) => {
504            info.push(Line::from(vec![
505                Span::styled("manifest: ", Style::default().fg(ACCENT)),
506                Span::raw(m.manifest_path.display().to_string()),
507            ]));
508            let mut scalars = Vec::new();
509            if !manifest.name.is_empty() {
510                scalars.push(format!("name {}", manifest.name));
511            }
512            if let Some(v) = &manifest.version {
513                scalars.push(format!("v{v}"));
514            }
515            if let Some(d) = &manifest.description {
516                scalars.push(d.clone());
517            }
518            if !scalars.is_empty() {
519                info.push(Line::from(Span::styled(
520                    scalars.join(" · "),
521                    Style::default().fg(MUTED),
522                )));
523            }
524            if !manifest.include.is_empty() {
525                info.push(Line::from(vec![
526                    Span::styled("include:  ", Style::default().fg(ACCENT)),
527                    Span::raw(manifest.include.join(", ")),
528                ]));
529            }
530            if manifest.provider.is_some() {
531                info.push(Line::from(vec![
532                    Span::styled("provider: ", Style::default().fg(ACCENT)),
533                    Span::raw("declared (read-only here)"),
534                ]));
535            }
536        }
537        (None, Some(err)) => {
538            info.push(Line::from(Span::styled(
539                format!("{err} — `:manifest` retries once the file is fixed"),
540                Style::default().fg(WARN),
541            )));
542        }
543        (None, None) => {
544            info.push(Line::from(Span::styled(
545                "no project manifest — adding a dep writes tamias.yaml + tamias.lock",
546                Style::default().fg(MUTED),
547            )));
548        }
549    }
550    // Deps table gets the rest of the frame; the info strip yields on
551    // short terminals.
552    let info_h = if area.height > 14 {
553        5.min(area.height / 3)
554    } else {
555        0
556    };
557    let (info_area, table_area) = if info_h > 0 {
558        let chunks = Layout::vertical([Constraint::Length(info_h), Constraint::Min(0)]).split(area);
559        (chunks[0], chunks[1])
560    } else {
561        (Rect::default(), area)
562    };
563    if info_h > 0 {
564        f.render_widget(
565            Paragraph::new(info).style(Style::default().fg(TEXT)),
566            info_area,
567        );
568    }
569    let deps: Vec<String> = m
570        .project
571        .as_ref()
572        .map(|p| p.deps.clone())
573        .unwrap_or_default();
574    if deps.is_empty() {
575        f.render_widget(
576            Paragraph::new("no deps — add refs under `deps:` in tamias.yaml")
577                .style(Style::default().fg(MUTED)),
578            table_area,
579        );
580        m.board_area = Rect::default();
581        m.viewport_top = 0;
582        return;
583    }
584    let header = Row::new(vec!["", "#", "REF"])
585        .style(Style::default().fg(ACCENT).add_modifier(Modifier::BOLD));
586    let rows: Vec<Row> = deps
587        .iter()
588        .enumerate()
589        .map(|(i, d)| {
590            Row::new(vec![
591                sel_cell(i, m.man_sel),

789                sel_cell(pos, m.sel),

792                Cell::from(rec.source_ref.clone()),
793                Cell::from(short_sha(&rec.digest)),
794                Cell::from(r.size.map_or("-".into(), human_size)),

808        .block(Block::default().borders(Borders::ALL).title(" installed "))
809        .style(Style::default().fg(TEXT))
810        .row_highlight_style(Style::default().bg(Color::Indexed(238)));
811    let mut state = ratatui::widgets::TableState::default().with_selected(Some(m.sel));
812    f.render_stateful_widget(table, area, &mut state);
813    note_table_area(m, area, state.offset());
814}

843/// Everything the package page renders, built from the [`Detail`]
844/// variant — owned `'static` lines so the borrow on `m.detail` ends
845/// before the frame's scroll offset is clamped. Layout follows the
846/// npm package-page anatomy: bold label + chips pinned at the top,
847/// then a scrollable body led by the full description (the old
848/// pinned ≤3-line strip clipped long text with no way to reach the
849/// rest), the `tamias install` command, then `── section ──` rules
850/// ordered by user value — deps/services/keywords + meta + versions —
851/// with provenance last.
852///
853/// `installed_ver` is the store's version label for the page's
854/// `(name, scope)` when one is installed
855/// ([`Model::installed_label_for`]) — Registry pages only; `None`
856/// elsewhere.
857///
858/// The Registry page's `── versions ──` section is fed by the
859/// background fetch [`Model::fetch_versions`] kicks off on open: a
860/// muted `versions …` placeholder shows while it's in flight, the
861/// section appears once the index read lands (index order, the page's
862/// own entry marked `← current`), and a fetch failure degrades to a
863/// muted line — never an error toast.
864///
865/// `versions_sel` is `Some(i)` while the versions-focus mode is
866/// active ([`Model::versions_focused`]) — row `i` gets the table's
867/// `Indexed(238)` row highlight plus an accent `›` marker, and the
868/// footer hint switches to the focus keys. It marks the Registry
869/// `── versions ──` list or the Installed `── installed versions ──`
870/// list (`installed_versions`, semver-desc store members — row 0 is
871/// the `· current` semver-max).
872///
873/// The fourth return value is the body line index of the focused
874/// version row (`None` unless focus is active and the list landed) —
875/// [`draw_detail`] converts it to display rows for
876/// [`scroll_to_show`] so the cursor row stays on screen past one
877/// page of versions.
878fn page_parts(
879    detail: &Detail,
880    registry_target: &str,
881    installed_ver: Option<&str>,
882    versions_sel: Option<usize>,
883    installed_versions: &[StoreEntry],
884) -> (String, Line<'static>, Vec<Line<'static>>, Option<usize>) {
885    let label = detail.label();
886    // Header — bold label then muted/bracketed chips; the trust chip
887    // is the one colored span (tier → [`trust_style`]).
888    let mut head = vec![Span::styled(
889        label.clone(),
890        Style::default()
891            .fg(Color::White)
892            .add_modifier(Modifier::BOLD),
893    )];
894    if let Some(kind) = detail.kind() {
895        head.push(chip(kind));
896    }
897    if let Some(scope) = detail.scope() {
898        head.push(chip(&format!("@{scope}")));
899    }
900    if let Some(license) = detail.license() {
901        head.push(chip(license));
902    }
903    if let Some(tier) = detail.trust() {
904        head.push(Span::styled(format!(" [{tier}]"), trust_style(&tier)));
905    }
906    if detail.sealed() {
907        head.push(chip("sealed"));
908    }
909    let head = Line::from(head);
910
911    let mut body: Vec<Line> = Vec::new();
912    // Description leads the scrollable body — full text, muted, one
913    // `Line` per source line (the body Paragraph does the wrapping,
914    // so nothing is clipped). It used to be a pinned ≤3-line strip
915    // whose overflow was unreachable.
916    if let Some(d) = detail.description() {
917        if !d.is_empty() {
918            for l in d.lines() {
919                body.push(Line::from(Span::styled(
920                    l.to_string(),
921                    Style::default().fg(MUTED),
922                )));
923            }
924            body.push(Line::from(""));
925        }
926    }
927    // Body index of the versions-focus row — set by the Registry
928    // arm below; `draw_detail` scrolls it into view.
929    let mut versions_row: Option<usize> = None;
930    // Install command — the ref an installed entry recorded, or the
931    // registry entry's resolved install ref (error inline on failure).
932    match detail {
933        Detail::Installed { entry, .. } => body.push(Line::from(vec![
934            Span::styled("install  ", Style::default().fg(ACCENT)),
935            Span::styled("tamias install ", Style::default().fg(MUTED)),
936            Span::raw(entry.record.source_ref.clone()),
937        ])),
938        Detail::Registry {
939            install_ref,
940            install_err,
941            ..
942        } => match (install_ref, install_err) {
943            (Some(reference), _) => body.push(Line::from(vec![
944                Span::styled("install  ", Style::default().fg(ACCENT)),
945                Span::styled("tamias install ", Style::default().fg(MUTED)),
946                Span::raw(reference.clone()),
947            ])),
948            (None, Some(err)) => body.push(Line::from(vec![
949                Span::styled("install  ", Style::default().fg(ACCENT)),
950                Span::styled("unavailable — ", Style::default().fg(WARN)),
951                Span::raw(format!("{err} — `:registry <name>` picks another target")),
952            ])),
953            (None, None) => {}
954        },
955    }
956
957    match detail {
958        Detail::Installed {
959            entry,
960            manifest,
961            manifest_err,
962        } => {
963            let rec = &entry.record;
964            if let Some(m) = manifest {
965                // deps — the manifest's direct dependency refs.
966                if !m.deps.is_empty() {
967                    body.push(Line::from(""));
968                    body.push(rule("deps"));
969                    for dep in &m.deps {
970                        body.push(Line::from(Span::raw(format!("  {dep}"))));
971                    }
972                }
973                // provider services — name + exec, the deploy surface.
974                if let Some(provider) = &m.provider {
975                    if !provider.services.is_empty() {
976                        body.push(Line::from(""));
977                        body.push(rule("services"));
978                        for s in &provider.services {
979                            body.push(Line::from(vec![
980                                Span::raw(format!("  {}", s.name)),
981                                Span::styled(format!("  {}", s.exec), Style::default().fg(MUTED)),
982                            ]));
983                        }
984                    }
985                }
986                // meta — the manifest's display fields the chips
987                // don't already carry.
988                let mut meta: Vec<Line> = Vec::new();
989                if !m.keywords.is_empty() {
990                    meta.push(kv("keywords", m.keywords.join(", ")));
991                }
992                if let Some(homepage) = &m.homepage {
993                    meta.push(kv("homepage", homepage.clone()));
994                }
995                if let Some(repository) = &m.repository {
996                    meta.push(kv("repository", repository.clone()));
997                }
998                if !m.authors.is_empty() {
999                    meta.push(kv("authors", m.authors.join(", ")));
1000                }
1001                if !m.include.is_empty() {
1002                    meta.push(kv("include", m.include.join(", ")));
1003                }
1004                if let Some(sealed) = &m.sealed {
1005                    meta.push(kv(
1006                        "sealed",
1007                        format!(
1008                            "{} — {} secret(s), ciphertext-only",
1009                            sealed.format,
1010                            sealed.files.len()
1011                        ),
1012                    ));
1013                }
1014                if !meta.is_empty() {
1015                    body.push(Line::from(""));
1016                    body.push(rule("meta"));
1017                    body.extend(meta);
1018                }
1019            } else if let Some(err) = manifest_err {
1020                body.push(Line::from(vec![
1021                    Span::styled("manifest ", Style::default().fg(WARN)),
1022                    Span::raw(format!(
1023                        "{err} — `:remove <name>` deletes it; reinstall repairs"
1024                    )),
1025                ]));
1026            }
1027            // installed versions — every store copy of this package,
1028            // semver-desc (`installed_versions` comes from the live
1029            // entries, so `:prune`/installs reshape it); row 0 is the
1030            // `· current` semver-max the Installed pane groups on.
1031            // `v` focuses the list for exact-version reinstall.
1032            if !installed_versions.is_empty() {
1033                body.push(Line::from(""));
1034                body.push(rule("installed versions"));
1035                let rows_start = body.len();
1036                versions_row = versions_sel
1037                    .filter(|&s| s < installed_versions.len())
1038                    .map(|s| rows_start + s);
1039                for (i, v) in installed_versions.iter().enumerate() {
1040                    // `record.version` when the install was versioned;
1041                    // unversioned copies show the sha12 the store
1042                    // label falls back to (same as `installed_labels`).
1043                    let ver = v.record.version.clone().unwrap_or_else(|| {
1044                        let sha = v
1045                            .record
1046                            .resolved
1047                            .strip_prefix("sha256:")
1048                            .unwrap_or(&v.record.resolved);
1049                        sha.chars().take(12).collect()
1050                    });
1051                    let current = i == 0;
1052                    let line = if versions_sel == Some(i) {
1053                        let mut spans = vec![
1054                            Span::styled("› ", Style::default().fg(ACCENT)),
1055                            Span::raw(ver),
1056                        ];
1057                        if current {
1058                            spans.push(Span::styled("  · current", Style::default().fg(MUTED)));
1059                        }
1060                        Line::from(spans).style(Style::default().bg(Color::Indexed(238)))
1061                    } else {
1062                        let mut spans = vec![Span::raw(format!("  {ver}"))];
1063                        if current {
1064                            spans.push(Span::styled("  · current", Style::default().fg(MUTED)));
1065                        }
1066                        Line::from(spans)
1067                    };
1068                    body.push(line);
1069                }
1070            }
1071            // provenance — last, per the package-page anatomy.
1072            body.push(Line::from(""));
1073            body.push(rule("provenance"));
1074            body.push(kv("source", rec.source_ref.clone()));
1075            body.push(kv("resolved", rec.resolved.clone()));
1076            body.push(kv("digest", rec.digest.clone()));
1077            // `installed_at` is unix seconds — render the same
1078            // relative label Home uses, not the raw epoch.
1079            body.push(kv(
1080                "installed",
1081                if rec.installed_at > 0 {
1082                    ago(rec.installed_at)
1083                } else {
1084                    "-".to_string()
1085                },
1086            ));
1087            body.push(kv("dir", entry.dir.display().to_string()));
1088            if let Some(signer) = &rec.signer {
1089                body.push(kv("signer", signer.clone()));
1090            }
1091            body.push(kv("trust", rec.trust.clone().unwrap_or_else(|| "-".into())));
1092        }
1093        Detail::Registry {
1094            entry,
1095            versions,
1096            versions_err,
1097            ..
1098        } => {
1099            // installed marker — the same (name, scope) is already in
1100            // the store, possibly at another version; sits right
1101            // under the install command.
1102            if let Some(ver) = installed_ver {
1103                let pkg = match &entry.scope {
1104                    Some(s) => format!("@{s}/{}", entry.name),
1105                    None => entry.name.clone(),
1106                };
1107                body.push(Line::from(vec![
1108                    Span::styled("installed  ", Style::default().fg(ACCENT)),
1109                    Span::styled(format!("{pkg}@{ver}"), Style::default().fg(MUTED)),
1110                ]));
1111            }
1112            // keywords — the index analogue of deps (search matches
1113            // them, so they double as discovery terms).
1114            if !entry.keywords.is_empty() {
1115                body.push(Line::from(vec![
1116                    Span::styled("keywords: ", Style::default().fg(ACCENT)),
1117                    Span::raw(entry.keywords.join(", ")),
1118                ]));
1119            }
1120            // meta — index-carried display fields (P-006).
1121            let mut meta: Vec<Line> = Vec::new();
1122            if let Some(homepage) = &entry.homepage {
1123                meta.push(kv("homepage", homepage.clone()));
1124            }
1125            if let Some(repository) = &entry.repository {
1126                meta.push(kv("repository", repository.clone()));
1127            }
1128            if !entry.authors.is_empty() {
1129                meta.push(kv("authors", entry.authors.join(", ")));
1130            }
1131            if !meta.is_empty() {
1132                body.push(Line::from(""));
1133                body.push(rule("meta"));
1134                body.extend(meta);
1135            }
1136            // versions — the background index fetch; pending shows a
1137            // subtle placeholder, the section renders once data lands
1138            // (a lone current version still lists — the header already
1139            // carries it, so no "no other versions" special-case).
1140            match (versions, versions_err) {
1141                (Some(list), _) => {
1142                    body.push(Line::from(""));
1143                    body.push(rule("versions"));
1144                    if list.is_empty() {
1145                        body.push(Line::from(Span::styled(
1146                            "  no versions indexed",
1147                            Style::default().fg(MUTED),
1148                        )));
1149                    }
1150                    // The focused row's body index — `draw_detail`
1151                    // scrolls it into view while versions focus is on.
1152                    let rows_start = body.len();
1153                    versions_row = versions_sel
1154                        .filter(|&s| s < list.len())
1155                        .map(|s| rows_start + s);
1156                    for (i, v) in list.iter().enumerate() {
1157                        let ver = v.version.clone().unwrap_or_else(|| "-".into());
1158                        // `file` is the per-version blob path — the
1159                        // unique match even if version strings repeat.
1160                        let current = v.file == entry.file;
1161                        let line = if versions_sel == Some(i) {
1162                            let mut spans = vec![
1163                                Span::styled("› ", Style::default().fg(ACCENT)),
1164                                Span::raw(ver),
1165                            ];
1166                            if current {
1167                                spans
1168                                    .push(Span::styled("  ← current", Style::default().fg(ACCENT)));
1169                            }
1170                            Line::from(spans).style(Style::default().bg(Color::Indexed(238)))
1171                        } else {
1172                            let mut spans = vec![Span::raw(format!("  {ver}"))];
1173                            if current {
1174                                spans
1175                                    .push(Span::styled("  ← current", Style::default().fg(ACCENT)));
1176                            }
1177                            Line::from(spans)
1178                        };
1179                        body.push(line);
1180                    }
1181                }
1182                (None, Some(err)) => body.push(Line::from(vec![
1183                    Span::styled("versions ", Style::default().fg(ACCENT)),
1184                    Span::styled(
1185                        format!("— {err} — reopen the page to retry"),
1186                        Style::default().fg(WARN),
1187                    ),
1188                ])),
1189                (None, None) => body.push(Line::from(vec![
1190                    Span::styled("versions ", Style::default().fg(ACCENT)),

1247/// Scroll offset that keeps body line `row` inside a `view_h`-line
1248/// window — scrolls the minimum amount (up when the row is above the
1249/// window, down when below); the caller still clamps to `max_scroll`.
1250/// `view_h == 0` leaves the offset alone.
1251fn scroll_to_show(row: usize, scroll: usize, view_h: usize) -> usize {
1252    if view_h == 0 {
1253        return scroll;
1254    }
1255    if row < scroll {
1256        row
1257    } else if row >= scroll + view_h {
1258        row + 1 - view_h
1259    } else {
1260        scroll
1261    }
1262}

1264/// Greedy word-wrap line count for `text` at `width` columns,
1265/// capped at `max` — the package page's pinned strips measure
1266/// themselves with this (`Paragraph::line_count` is unstable in
1267/// ratatui 0.30). Overlong words count as filling their line; the
1268/// estimate only sizes a strip, never clips content.
1269fn wrapped_lines(text: &str, width: u16, max: usize) -> usize {
1270    let width = (width as usize).max(1);
1271    let mut lines = 1usize;
1272    let mut col = 0usize;
1273    for word in text.split_whitespace() {
1274        let ww = Span::raw(word).width();
1275        if col == 0 {
1276            col = ww;
1277        } else if col + 1 + ww <= width {
1278            col += 1 + ww;
1279        } else {
1280            lines += 1;
1281            col = ww;
1282        }
1283        if lines >= max {
1284            return max;
1285        }
1286    }
1287    lines.min(max)
1288}

1290/// Wrapped display-line count for `lines` at `width` columns —
1291/// `Paragraph`'s own `line_count` (the `unstable-rendered-line-info`
1292/// feature), i.e. the exact `WordWrapper` math the render path uses
1293/// with `Wrap { trim: false }`. Scroll offsets are display rows, so
1294/// max-scroll and cursor-row math must count these, never source
1295/// `Line`s — a hand-rolled estimate miscounts hard-wrapped tokens
1296/// (long digests, URLs) that `WordWrapper` splits mid-word.
1297fn wrapped_count(lines: &[Line], width: u16) -> usize {
1298    Paragraph::new(lines.to_vec())
1299        .wrap(Wrap { trim: false })
1300        .line_count(width)
1301}

1303/// The `Enter` package page — npm-inspired anatomy inside the
1304/// bordered block. The header (≤3 wrapped lines) stays pinned; the
1305/// body — description, install command, sections, provenance —
1306/// scrolls with `j`/`k` via `Model::detail_scroll`, counted in
1307/// *display* rows (the Paragraph wraps).
1308fn draw_detail(f: &mut Frame, m: &mut Model, area: Rect) {
1309    let Some(detail) = m.detail.as_ref() else {
1310        return;
1311    };
1312    // Registry pages mark an existing install of the same
1313    // (name, scope) — any version counts.
1314    let installed_ver = match detail {
1315        Detail::Registry { entry, .. } => {
1316            m.installed_label_for(&entry.name, entry.scope.as_deref())
1317        }
1318        _ => None,
1319    };
1320    // The Installed page's version list — every store member of the
1321    // group, recomputed live so `:prune`/installs reshape it.
1322    let installed_versions = match detail {
1323        Detail::Installed { .. } => m.installed_page_versions(),
1324        _ => Vec::new(),
1325    };
1326    let (title, head, body, versions_row) = page_parts(
1327        detail,
1328        &m.reg_target_label,
1329        installed_ver.as_deref(),
1330        m.versions_focused().then_some(m.versions_sel),
1331        &installed_versions,
1332    );
1333    let block = Block::default()
1334        .borders(Borders::ALL)
1335        .title(format!(" {title} "));
1336    let inner = block.inner(area);
1337    f.render_widget(block, area);
1338    if inner.width == 0 || inner.height == 0 {
1339        m.board_area = Rect::default();
1340        m.viewport_top = 0;
1341        return;
1342    }
1343    // Pinned strip — the header wraps to at most 3 lines; the body
1344    // below carries everything else and scrolls.
1345    let header_text: String = head.spans.iter().map(|s| s.content.as_ref()).collect();
1346    let header_h = wrapped_lines(&header_text, inner.width, 3).min(inner.height as usize) as u16;
1347    let body_area = Rect {
1348        y: inner.y + header_h,
1349        height: inner.height.saturating_sub(header_h),
1350        ..inner
1351    };
1352    let header_area = Rect {
1353        height: header_h,
1354        ..inner
1355    };
1356    f.render_widget(
1357        Paragraph::new(head)
1358            .style(Style::default().fg(TEXT))
1359            .wrap(Wrap { trim: true }),
1360        header_area,
1361    );
1362    // Versions focus keeps its cursor row on screen. `versions_row`
1363    // is a body *source*-line index while `detail_scroll` counts
1364    // display rows — convert both ends of the row through
1365    // [`wrapped_count`] (prefix counts give the row's first/last
1366    // display row) and aim the scroll at the whole row, minimal
1367    // scroll first.
1368    if let Some(row) = versions_row {
1369        let top = wrapped_count(&body[..row], body_area.width);
1370        let bottom = wrapped_count(&body[..=row], body_area.width).saturating_sub(1);
1371        let view_h = body_area.height as usize;
1372        m.detail_scroll = scroll_to_show(bottom, m.detail_scroll, view_h);
1373        m.detail_scroll = scroll_to_show(top, m.detail_scroll, view_h);
1374    }
1375    // Clamp the scroll offset to the body's *wrapped* height, then
1376    // render. The max is recorded for `G` — the model's
1377    // jump-to-bottom target.
1378    let para = Paragraph::new(body)
1379        .style(Style::default().fg(TEXT))
1380        .wrap(Wrap { trim: false });
1381    let max_scroll = para
1382        .line_count(body_area.width)
1383        .saturating_sub(body_area.height as usize);
1384    m.detail_scroll = m.detail_scroll.min(max_scroll);
1385    m.detail_max_scroll = max_scroll;
1386    f.render_widget(para.scroll((m.detail_scroll as u16, 0)), body_area);
1387    m.board_area = Rect::default();
1388    m.viewport_top = 0;
1389}

1736/// `s` clipped to `max` display columns with a trailing `…` — the
1737/// status bar's right-side hint yields to the left status text.
1738fn fit_width(s: &str, max: usize) -> String {
1739    if max == 0 {
1740        return String::new();
1741    }
1742    if Line::raw(s).width() <= max {
1743        return s.to_string();
1744    }
1745    let budget = max - 1; // room for the ellipsis
1746    let mut out = String::new();
1747    let mut w = 0usize;
1748    for ch in s.chars() {
1749        let cw = Span::raw(ch.to_string()).width();
1750        if w + cw > budget {
1751            break;
1752        }
1753        w += cw;
1754        out.push(ch);
1755    }
1756    out.push('…');
1757    out
1758}
