1//! tamias-driver-bupha — dual-surface driver for bupha's CI state.
2//!
3//! One binary, two consumers, disjoint argv:
4//!
5//! **pavo view driver** (verbs take no `--dir`/`--name`):
6//!
7//! - `manifest` — print the [`pavo_protocol::DriverManifest`] JSON
8//!   (discovery probe; never touches bupha state).
9//! - `capabilities` — merged JSON advertising BOTH surfaces: pavo's
10//!   `{"capabilities": ["pavo"], "verbs": […"serve"…]}` shape plus
11//!   tamias's `"types": ["bupha"]` and the provider verbs in `verbs`.
12//!   pavo's discovery greps `capabilities`/`verbs` for
13//!   `"pavo"`/`"serve"`; tamias's greps `types` for the content kind
14//!   and warns on verbs missing from `verbs` — one payload satisfies
15//!   both.
16//! - `serve` — ndjson JSON-RPC on stdio (see `bupha_driver::serve`).
17//!
18//! **tamias content-type driver** (`type: bupha` packages — tamias
19//! always passes `--dir <payload> --name <pkg> [--config <f>]`, so
20//! these verbs are unreachable from pavo's bare probes):
21//!
22//! - `install` — no-op (the operator config is `deploy`'s job; install
23//!   only sees the evictable fetch-cache payload).
24//! - `host` — engine-host mode: run the watcher pipeline
25//!   (`bupha-watch`) in-process against the deployed operator
26//!   config — what provider units exec (`tamias-driver-bupha host
27//!   --config ~/.config/bupha/bundles/<name>.toml`). Blocks until
28//!   the sources stop; `--dir`/`--name` accepted per the tamias
29//!   dispatch contract (`--name` supplies the default
30//!   `bundles/<name>.toml` path when `--config` is absent).
31//!   **Standalone/dev mode** (ADR-0007): the trigger listener this
32//!   hosts is the fallback ingress — production webhook ingress is
33//!   a peregrin campaign listener enqueuing `ci-job` via
34//!   `ci.enqueue`.
35//! - `worker` — the `ci-job` consumption loop (P-002 §4.1): polls
36//!   `kind="ci-job"` entities on a hypha/peregrin node
37//!   (`[worker] node` / `--node` / `BUPHA_WORKER_NODE`, credential
38//!   via `[worker] node_token_env` / `--node-token-env` /
39//!   `BUPHA_NODE_TOKEN`), `ci.claim`s pending jobs this box's
40//!   `[worker] labels` cover, runs them through the same
41//!   `bupha-watch` pipeline, `ci.complete`s the result. The
42//!   **second service** a fleet deployment runs — worker boxes
43//!   execute jobs the entity plane hands out.
44//! - `deploy` — merge the payload's `fleet.toml` with
45//!   `tamias.yaml provider.ingress[]` (`tunnel.public_url`,
46//!   `trigger.listen`/`kind`) and write
47//!   `~/.config/bupha/bundles/<name>.toml` — the file the provider
48//!   unit's `host` verb consumes.
49//! - `uninstall` — remove that file.
50//! - `hooks register|uninstall [--dry-run] [--api-base <url>]
51//!   [--token-env <NAME>]` — upsert/remove the GitHub webhooks
52//!   `provider.ingress[].hooks[]` declare, via `bupha_github`'s REST
53//!   path. Secrets resolve by NAME (env → cuora → keyring →
54//!   `~/.config/bupha/env`; cuora canonical per T-056); values are
55//!   never printed.
56//!   **Standalone mode** (ADR-0007): registers hooks at the bupha
57//!   listener — production ingress is peregrin campaigns, so use
58//!   this only for deployments with no peregrin.
59//! - `health` — JSON probe: webhook `ping` per ingress
60//!   `listen`+`path`, else `systemctl --user is-active` per unit.
61//! - `status` — JSON per-unit `is-active`/`is-enabled`.
62//! - `logs [--lines N]` — `journalctl --user` over the provider units.
63//! - `config-schema` — JSON Schema (draft 2020-12) of the operator
64//!   config `deploy` writes (tamias's reserved verb).
65//!
66//! See `bupha_driver::ops` for the implementation and
67//! `bupha_driver::data` for the pavo read plane.
68
69use bupha_driver as drv;
70
71fn main() {
72    let mut args = std::env::args().skip(1);
73    match args.next().as_deref() {
74        Some("manifest") => {
75            println!(
76                "{}",
77                serde_json::to_string_pretty(&drv::manifest_value()).unwrap_or_default()
78            );
79        }
80        Some("capabilities") => {
81            println!(
82                "{}",
83                serde_json::to_string_pretty(&drv::capabilities_value()).unwrap_or_default()
84            );
85        }
86        Some("serve") => std::process::exit(drv::serve()),
87        // Tamias verbs — always invoked with `--dir`/`--name` by the
88        // tamias dispatch contract; pavo never passes those flags, so
89        // subcommand-first dispatch keeps the surfaces disjoint.
90        Some(
91            verb @ ("install" | "deploy" | "uninstall" | "hooks" | "health" | "status" | "logs"
92            | "host" | "worker" | "config-schema"),
93        ) => {
94            std::process::exit(drv::ops::run(verb, &args.collect::<Vec<_>>()));
95        }
96        _ => {
97            eprintln!("usage: tamias-driver-bupha manifest|capabilities|serve");
98            eprintln!("       tamias-driver-bupha <install|deploy|uninstall|hooks|health|status|logs|host|worker|config-schema> --dir <d> --name <n> [--config <c>]");
99            std::process::exit(1);
100        }
101    }
102}
