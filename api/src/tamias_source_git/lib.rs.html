19/// A parsed git ref.
20#[derive(Clone, Debug, PartialEq, Eq)]
21pub struct GitSpec {
22    /// Normalized clone URL (`org/repo` → `https://github.com/org/repo`).
23    pub url: String,
24    /// In-repo path after `//` (or the post-two-segment remainder).
25    pub subdir: Option<String>,
26    /// `#<branch|tag|sha>` — the version pin.
27    pub rev: Option<String>,
28}

30impl RefKind for GitRefKind {
31    fn claims(&self, input: &str) -> bool {
32        input.starts_with("git:")
33    }
34
35    fn parse(&self, input: &str) -> Result<Spec, Error> {
36        parse_git(input.strip_prefix("git:").unwrap_or_default())
37    }
38}

40/// `git:<rest>` — `#rev` off the tail, `//subdir` out of the middle, then
41/// clone-URL normalization.
42fn parse_git(rest: &str) -> Result<Spec, Error> {
43    if rest.is_empty() {
44        return Err(Error::Failed("git: ref is empty".to_string()));
45    }
46    let (head, rev) = match rest.split_once('#') {
47        Some((head, rev)) => {
48            if rev.is_empty() {
49                return Err(Error::Failed("empty ref after '#'".to_string()));
50            }
51            if rev.contains('#') {
52                return Err(Error::Failed(format!("malformed ref '#{rev}'")));
53            }
54            (head, Some(rev.to_string()))
55        }
56        None => (rest, None),
57    };
58    if head.is_empty() {
59        return Err(Error::Failed("git: ref missing the repo".to_string()));
60    }
61    let (repo, subdir) = split_subdir(head)?;
62    Ok(Box::new(GitSpec {
63        url: normalize_clone_url(repo),
64        subdir,
65        rev,
66    }))
67}
68
69/// Pull the `//subdir` out of `head`; fall back to the two-segment rule.
70fn split_subdir(head: &str) -> Result<(&str, Option<String>), Error> {
71    let scan = head.find("://").map_or(0, |i| i + 3);
72    if let Some(i) = head[scan..].find("//") {
73        let repo = &head[..scan + i];
74        let subdir = head[scan + i + 2..].trim_end_matches('/');
75        if subdir.is_empty() || subdir == "." {
76            return Ok((repo, None));
77        }
78        if subdir.starts_with('/') || subdir.split('/').any(|c| c == "..") {
79            return Err(Error::Failed(format!("bad in-repo path '//{subdir}'")));
80        }
81        return Ok((repo, Some(subdir.to_string())));
82    }
83    if head.contains("://")
84        || is_scp_style(head)
85        || head.starts_with('/')
86        || head.starts_with('.')
87        || head.starts_with('~')
88    {
89        return Ok((head, None));
90    }
91    let Some(first_slash) = head.find('/') else {
92        return Ok((head, None));
93    };
94    let Some(second_slash) = head[first_slash + 1..].find('/') else {
95        return Ok((head, None));
96    };
97    let repo_end = first_slash + 1 + second_slash;
98    let subdir = head[repo_end + 1..].trim_end_matches('/');
99    if subdir.is_empty() || subdir == "." {
100        return Ok((&head[..repo_end], None));
101    }
102    if subdir.split('/').any(|c| c == "..") {
103        return Err(Error::Failed(format!("bad in-repo path '/{subdir}'")));
104    }
105    Ok((&head[..repo_end], Some(subdir.to_string())))
106}

108/// scp-style ssh — `git@host:path`, `host:path`. The `:` lands before the
109/// first `/` (a `://` was already ruled out by the caller).
110fn is_scp_style(head: &str) -> bool {
111    head.split('/')
112        .next()
113        .is_some_and(|first| first.contains(':'))
114}
115
116/// Bare shorthand → a clone URL: `org/repo` → github.com,
117/// `host/org/repo` → `https://host/…`; URLs and scp pass through.
118fn normalize_clone_url(repo: &str) -> String {
119    if repo.contains("://")
120        || is_scp_style(repo)
121        || repo.starts_with('/')
122        || repo.starts_with('.')
123        || repo.starts_with('~')
124    {
125        return repo.to_string();
126    }
127    let mut segments = repo.split('/');
128    let first = segments.next().unwrap_or(repo);
129    let two_segments = segments.clone().count() == 1;
130    if two_segments && !first.contains('.') {
131        format!("https://github.com/{repo}")
132    } else {
133        format!("https://{repo}")
134    }
135}

137/// Run `git` in `cwd`, returning stdout (or the stderr on failure).
138fn run_git(cwd: &Path, args: &[&str]) -> Result<String, Error> {
139    let out = Command::new("git")
140        .current_dir(cwd)
141        .args(args)
142        .output()
143        .map_err(|e| Error::Failed(format!("git {}: {e}", args.join(" "))))?;
144    if !out.status.success() {
145        return Err(Error::Failed(format!(
146            "git {}: {}",
147            args.join(" "),
148            String::from_utf8_lossy(&out.stderr).trim()
149        )));
150    }
151    Ok(String::from_utf8_lossy(&out.stdout).to_string())
152}

154/// 40 lowercase-hex chars — already fully pinned, no `ls-remote`.
155fn is_full_sha(rev: &str) -> bool {
156    rev.len() == 40
157        && rev
158            .bytes()
159            .all(|b| b.is_ascii_hexdigit() && !b.is_ascii_uppercase())
160}
161
162/// A hex prefix (7..40) resolves at fetch — the clone carries the object.
163fn is_hex_prefix(rev: &str) -> bool {
164    rev.len() >= 7
165        && rev.len() < 40
166        && rev
167            .bytes()
168            .all(|b| b.is_ascii_hexdigit() && !b.is_ascii_uppercase())
169}

171/// `~`/`/`-leading clone paths expand like fs refs.
172fn expand_url(url: &str) -> Result<String, Error> {
173    if !(url.starts_with('~') || url.starts_with('/')) {
174        return Ok(url.to_string());
175    }
176    expand_path(Path::new(url))
177        .to_str()
178        .map(str::to_string)
179        .ok_or_else(|| Error::Failed(format!("non-UTF-8 clone path '{url}'")))
180}
181
182/// `ls-remote` stdout → `(refname → sha)`.
183fn parse_ls_remote(out: &str) -> Vec<(String, String)> {
184    out.lines()
185        .filter_map(|line| {
186            let (sha, refname) = line.split_once('\t')?;
187            Some((refname.trim().to_string(), sha.trim().to_string()))
188        })
189        .collect()
190}
191
192fn resolve_remote(url: &str, rev: Option<&str>) -> Result<String, Error> {
193    let out = run_git(Path::new("."), &["ls-remote", url])?;
194    let refs = parse_ls_remote(&out);
195    let pick = |name: &str| refs.iter().find(|(r, _)| r == name).map(|(_, s)| s.clone());
196    match rev {
197        Some(rev) => {
198            for cand in [
199                format!("refs/tags/{rev}^{{}}"),
200                format!("refs/tags/{rev}"),
201                format!("refs/heads/{rev}"),
202            ] {
203                if let Some(sha) = pick(&cand) {
204                    return Ok(sha);
205                }
206            }
207            Err(Error::Failed(format!("ref '{rev}' not found on '{url}'")))
208        }
209        None => pick("HEAD").ok_or_else(|| Error::Failed(format!("no HEAD on '{url}'"))),
210    }
211}

225struct GitPlan {
226    url: String,
227    subdir: Option<String>,
228    rev: Option<String>,
229    sha: Option<String>,
230}
231
232impl Plan for GitPlan {
233    fn prefetch_identity(&self) -> Option<(String, String)> {
234        self.sha
235            .as_ref()
236            .map(|sha| (format!("git-{sha}"), sha.clone()))
237    }
238}
239
240impl Source for GitSource {
241    fn accepts(&self, spec: &dyn Any) -> bool {
242        spec.is::<GitSpec>()
243    }
244
245    fn plan(&self, spec: &dyn Any) -> Result<Box<dyn Plan>, Error> {
246        let GitSpec { url, subdir, rev } = spec
247            .downcast_ref::<GitSpec>()
248            .ok_or_else(|| Error::Failed("GitSource only plans git refs".to_string()))?;
249        let url = expand_url(url)?;
250        let sha = match rev.as_deref() {
251            Some(rev) if is_full_sha(rev) => Some(rev.to_string()),
252            Some(rev) if is_hex_prefix(rev) => None,
253            other => Some(resolve_remote(&url, other)?),
254        };
255        Ok(Box::new(GitPlan {
256            url,
257            subdir: subdir.clone(),
258            rev: rev.clone(),
259            sha,
260        }))
261    }
262
263    fn fetch(&self, plan: &dyn Plan, dest_dir: &Path) -> Result<Fetched, Error> {
264        let GitPlan {
265            url,
266            subdir,
267            rev,
268            sha,
269        } = plan
270            .downcast_ref::<GitPlan>()
271            .ok_or_else(|| Error::Failed("GitSource only fetches git plans".to_string()))?;
272        std::fs::create_dir_all(dest_dir)
273            .map_err(Error::io(format!("create {}", dest_dir.display())))?;
274        let clone = dest_dir.join(".clone");
275        let clone_str = clone.to_string_lossy().to_string();
276        run_git(dest_dir, &["clone", "--quiet", url, &clone_str])?;
277        let want = sha.as_deref().or(rev.as_deref()).unwrap_or("HEAD");
278        let want_commit = format!("{want}^{{commit}}");
279        let sha = run_git(&clone, &["rev-parse", "--verify", &want_commit])?
280            .trim()
281            .to_string();
282        run_git(&clone, &["checkout", "--quiet", &sha])?;
283        let payload = subdir
284            .as_ref()
285            .map_or_else(|| clone.clone(), |s| clone.join(s));
286        if !payload.is_dir() {
287            return Err(Error::Failed(format!(
288                "subdir '{}' not in {url}@{sha}",
289                subdir.as_deref().unwrap_or_default()
290            )));
291        }
292        copy_tree(&payload, dest_dir)?;
293        std::fs::remove_dir_all(&clone).map_err(Error::io(format!("clear {}", clone.display())))?;
294        Ok(Fetched {
295            dir: dest_dir.to_path_buf(),
296            resolved: sha.clone(),
297            content_key: format!("git-{sha}"),
298        })
299    }
300}

328    #[test]
329    fn grammar_plan_fetch_and_errors() {
330        let refs = Refs::new().with(Arc::new(GitRefKind));
331        let sources = Sources::new().with(Arc::new(GitSource::new()));
332
333        // grammar
334        let spec = refs
335            .parse("git:9aia/peregrin//libraries/ci-gate#v0.2.0")
336            .unwrap();
337        let GitSpec { url, subdir, rev } = *spec.downcast::<GitSpec>().unwrap();
338        assert_eq!(url, "https://github.com/9aia/peregrin");
339        assert_eq!(subdir.as_deref(), Some("libraries/ci-gate"));
340        assert_eq!(rev.as_deref(), Some("v0.2.0"));
341        assert_eq!(
342            *refs
343                .parse("git:org/repo")
344                .unwrap()
345                .downcast::<GitSpec>()
346                .unwrap(),
347            GitSpec {
348                url: "https://github.com/org/repo".into(),
349                subdir: None,
350                rev: None
351            }
352        );
353        assert!(refs.parse("git:").is_err());
354        assert!(refs.parse("git:x/y#").is_err());
355        assert!(refs.parse("git:x/y#a#b").is_err());
356        assert!(refs.parse("git:x/y//a/../b").is_err());
357        assert!(GitRefKind.claims("git:x"));
358        assert!(!GitRefKind.claims("./x"));
359
360        // a real repo → plan pins the sha; fetch clones
361        let repo = tempfile::tempdir().unwrap();
362        std::fs::write(repo.path().join("tamias.yaml"), "name: g\nversion: 1.0.0\n").unwrap();
363        std::fs::write(repo.path().join("a.txt"), "a").unwrap();
364        git(repo.path(), &["init", "-q", "-b", "main"]);
365        git(repo.path(), &["add", "-A"]);
366        git(repo.path(), &["commit", "-q", "-m", "i"]);
367        let url = format!("git:file://{}#main", repo.path().display());
368        let spec = refs.parse(&url).unwrap();
369        let planned = sources.plan(&*spec).unwrap();
370        assert!(planned.prefetch_identity().unwrap().0.starts_with("git-"));
371        let dest = tempfile::tempdir().unwrap();
372        let fetched = planned.fetch(dest.path()).unwrap();
373        assert!(fetched.content_key.starts_with("git-"));
374        assert!(dest.path().join("a.txt").is_file());
375
376        // a full-sha rev needs no ls-remote
377        let sha = run_git(&repo.path().join("."), &["rev-parse", "HEAD"]).unwrap();
378        let spec = refs
379            .parse(&format!(
380                "git:file://{}#{}",
381                repo.path().display(),
382                sha.trim()
383            ))
384            .unwrap();
385        assert!(sources.plan(&*spec).is_ok());
386
387        // errors: unknown rev / bad subdir / dead remote
388        let spec = refs
389            .parse(&format!("git:file://{}#nosuchrev", repo.path().display()))
390            .unwrap();
391        assert!(sources.plan(&*spec).is_err());
392        let spec = refs
393            .parse(&format!("git:file://{}//nope#main", repo.path().display()))
394            .unwrap();
395        let planned = sources.plan(&*spec).unwrap();
396        let d2 = tempfile::tempdir().unwrap();
397        assert!(planned.fetch(d2.path()).is_err());
398        let spec = refs.parse("git:file:///nonexistent-xyz-999#main").unwrap();
399        assert!(sources.plan(&*spec).is_err());
400
401        // foreign spec / plan
402        let source = GitSource::new();
403        assert!(!source.accepts(&1u8));
404        assert!(source.plan(&1u8).is_err());
405        assert!(source.fetch(&ForeignPlan, d2.path()).is_err());
406    }
