43impl Registry for LocalRegistry {
44    fn entries(&self) -> Result<Vec<RegistryEntry>, Error> {
45        let path = self.index_path();
46        if !path.is_file() {
47            return Ok(Vec::new());
48        }
49        let text =
50            fs::read_to_string(&path).map_err(Error::io(format!("read {}", path.display())))?;
51        serde_json::from_str(&text)
52            .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))
53    }
54
55    fn publish(&self, payload_root: &Path, manifest: &Manifest) -> Result<RegistryEntry, Error> {
56        // The manifest may not have been validated upstream — every field
57        // embedded in the blob path must be a single safe component.
58        if !is_valid_name(&manifest.name) {
59            return Err(Error::Failed(format!(
60                "unsafe package name '{}'",
61                manifest.name
62            )));
63        }
64        if let Some(scope) = &manifest.scope {
65            if !is_valid_name(scope) {
66                return Err(Error::Failed(format!("unsafe scope '{scope}'")));
67            }
68        }
69        let scoped = match &manifest.scope {
70            Some(scope) => format!("@{scope}/{}", manifest.name),
71            None => manifest.name.clone(),
72        };
73        let version = manifest
74            .version
75            .clone()
76            .unwrap_or_else(|| "0.0.0".to_string());
77        if !is_valid_version(&version) {
78            return Err(Error::Failed(format!(
79                "unsafe version '{version}' — expected [A-Za-z0-9][A-Za-z0-9._+-]* with no '..'"
80            )));
81        }
82        let rel = format!("{scoped}/{version}.tgz");
83        let out = self.root.join(&rel);
84        let parent = out.parent().unwrap_or(Path::new("."));
85        fs::create_dir_all(parent).map_err(Error::io(format!("create {}", parent.display())))?;
86        // Blob lands atomically — a crash mid-pack leaves a stray
87        // `*.tmp`, never a truncated `.tgz` at the canonical path.
88        pack_tgz_atomic(payload_root, &out)?;
89        let entry = RegistryEntry {
90            name: manifest.name.clone(),
91            scope: manifest.scope.clone(),
92            version: manifest.version.clone(),
93            kind: manifest.kind.clone(),
94            digest: dir_digest(payload_root)?,
95            file: rel,
96        };
97        // The index read-modify-write is serialized across processes —
98        // concurrent publishes can't lose each other's entries.
99        fs::create_dir_all(&self.root)
100            .map_err(Error::io(format!("create {}", self.root.display())))?;
101        let _lock = IndexLock::acquire(&self.root)?;
102        let mut entries = self.entries()?;
103        entries.retain(|e| {
104            !(e.name == entry.name && e.scope == entry.scope && e.version == entry.version)
105        });
106        entries.push(entry.clone());
107        entries
108            .sort_by(|a, b| (&a.scope, &a.name, &a.version).cmp(&(&b.scope, &b.name, &b.version)));
109        let text = serde_json::to_string_pretty(&entries)
110            .map_err(|e| Error::Failed(format!("serialize index: {e}")))?;
111        write_atomic(&self.index_path(), text.as_bytes())?;
112        Ok(entry)
113    }
114}

116/// `index.lock` — an atomic-create (`create_new`) lockfile serializing
117/// the registry index read-modify-write. A lock older than
118/// [`STALE_AFTER`] is treated as abandoned (holder crashed) and stolen;
119/// `create_new` still guarantees a single winner of the steal race.
120/// Residual: a holder legitimately keeping the lock past
121/// [`STALE_AFTER`] (a multi-minute index update) could have its lock
122/// stolen — index updates here are sub-second, so this is theoretical.
123struct IndexLock {
124    path: PathBuf,
125}
126
127/// A lockfile untouched for this long is considered stale.
128const STALE_AFTER: Duration = Duration::from_secs(30);
129/// How long `acquire` waits for a live holder before giving up.
130const LOCK_TIMEOUT: Duration = Duration::from_secs(10);
131
132impl IndexLock {
133    /// Take `root/index.lock`, waiting out live holders.
134    ///
135    /// # Errors
136    /// Fails on I/O errors or when the lock stays held past
137    /// [`LOCK_TIMEOUT`].
138    fn acquire(root: &Path) -> Result<Self, Error> {
139        let path = root.join("index.lock");
140        let deadline = Instant::now() + LOCK_TIMEOUT;
141        loop {
142            match fs::OpenOptions::new()
143                .write(true)
144                .create_new(true)
145                .open(&path)
146            {
147                Ok(mut f) => {
148                    use std::io::Write;
149                    let _ = writeln!(f, "{}", std::process::id());
150                    return Ok(Self { path });
151                }
152                Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => {
153                    if lock_stale(&path) {
154                        let _ = fs::remove_file(&path);
155                        continue;
156                    }
157                    if Instant::now() >= deadline {
158                        return Err(Error::Failed(format!(
159                            "registry index locked ({})",
160                            path.display()
161                        )));
162                    }
163                    std::thread::sleep(Duration::from_millis(20));
164                }
165                Err(e) => return Err(Error::io(format!("lock {}", path.display()))(e)),
166            }
167        }
168    }
169}
170
171impl Drop for IndexLock {
172    fn drop(&mut self) {
173        let _ = fs::remove_file(&self.path);
174    }
175}

177/// Whether `path`'s mtime is older than [`STALE_AFTER`] — or unreadable,
178/// in which case removing it is harmless (the create race decides).
179fn lock_stale(path: &Path) -> bool {
180    fs::metadata(path)
181        .and_then(|m| m.modified())
182        .map(|t| t.elapsed().is_ok_and(|age| age > STALE_AFTER))
183        .unwrap_or(true)
184}

186/// Largest PUT body the dev server accepts — matches the fetch-side
187/// `MAX_TARBALL` convention; bigger blobs can't be installed anyway.
188const MAX_PUT: u64 = 512 * 1024 * 1024;
189
190/// Per-read timeout on a connection — an idle client can't stall the
191/// single-threaded accept loop (each `read` gets this long).
192const READ_TIMEOUT: Duration = Duration::from_secs(30);

194/// Env var that turns on bearer auth for the dev server — mirrors the
195/// client's `TAMIAS_REGISTRY_TOKEN`. Unset = open (dev default).
196const SERVE_TOKEN_ENV: &str = "TAMIAS_REGISTRY_SERVE_TOKEN";

198/// Serve `root` over a tiny blocking HTTP surface until the process stops.
199/// `GET /<key>` reads a blob; `PUT /<key>` writes one. The registry root is
200/// the blob namespace (paths sanitized — no traversal).
201///
202/// **Dev surface, not a real registry.** Anyone who can reach the port can
203/// read and overwrite every blob — including `index.json` (the `PUT /`
204/// sanitize maps the root to it, and `HttpRegistry` publishes through this
205/// protocol, so it stays allowed). Bind loopback, or set
206/// `TAMIAS_REGISTRY_SERVE_TOKEN` to require `Authorization: Bearer <token>`.
207///
208/// # Errors
209/// Propagates the bind failure; per-request errors are logged, not fatal.
210pub fn serve_http(addr: &str, root: &Path) -> Result<(), Error> {
211    let token = std::env::var(SERVE_TOKEN_ENV)
212        .ok()
213        .filter(|t| !t.is_empty());
214    serve_http_token(addr, root, token.as_deref())
215}

217/// [`serve_http`] with an explicit bearer token — `Some` requires
218/// `Authorization: Bearer <token>` on every request. Split out so tests
219/// don't race on the process env.
220fn serve_http_token(addr: &str, root: &Path, token: Option<&str>) -> Result<(), Error> {
221    let listener = std::net::TcpListener::bind(addr).map_err(Error::io(format!("bind {addr}")))?;
222    fs::create_dir_all(root).map_err(Error::io(format!("create {}", root.display())))?;
223    eprintln!(
224        "tamias registry serving {} at http://{addr}/ — dev server: \
225         unauthenticated blob store (set {SERVE_TOKEN_ENV} for bearer auth)",
226        root.display()
227    );
228    // `--addr 0.0.0.0` (or any non-loopback bind) exposes the writable
229    // store to the network — say so loudly.
230    let exposed = addr
231        .to_socket_addrs()
232        .map(|mut addrs| addrs.any(|a| !a.ip().is_loopback()))
233        .unwrap_or(false);
234    if exposed {
235        eprintln!(
236            "warning: registry is bound to a non-loopback address ({addr}) — \
237             anyone who can reach it can read/overwrite all blobs"
238        );
239    }
240    for stream in listener.incoming() {
241        match stream {
242            Ok(mut stream) => {
243                if let Err(e) = handle(&mut stream, root, token) {
244                    eprintln!("tamias registry: {e}");
245                }
246            }
247            Err(e) => eprintln!("tamias registry: accept: {e}"),
248        }
249    }
250    Ok(())
251}

253fn handle(stream: &mut std::net::TcpStream, root: &Path, token: Option<&str>) -> Result<(), Error> {
254    use std::io::{BufRead, BufReader, Read};
255
256    // Bound every read on this connection — headers AND body — so an
257    // idle or slow-loris client can't park the accept loop.
258    stream
259        .set_read_timeout(Some(READ_TIMEOUT))
260        .map_err(Error::io("set read timeout"))?;
261    let clone = stream.try_clone().map_err(Error::io("clone stream"))?;
262    let mut reader = BufReader::new(clone);
263    let mut request_line = String::new();
264    reader
265        .read_line(&mut request_line)
266        .map_err(Error::io("read request"))?;
267    let mut parts = request_line.split_whitespace();
268    let method = parts.next().unwrap_or("").to_string();
269    let raw_path = parts.next().unwrap_or("/").to_string();
270    let mut content_length = 0u64;
271    let mut authorization = String::new();
272    loop {
273        let mut line = String::new();
274        reader
275            .read_line(&mut line)
276            .map_err(Error::io("read header"))?;
277        if line == "\r\n" || line == "\n" || line.is_empty() {
278            break;
279        }
280        // Match the header name case-insensitively but keep the VALUE
281        // verbatim — bearer tokens are case-sensitive.
282        let lower = line.to_ascii_lowercase();
283        if let Some(v) = lower.strip_prefix("content-length:") {
284            content_length = v.trim().parse().unwrap_or(0);
285        } else if lower.starts_with("authorization:") {
286            authorization = line["authorization:".len()..].trim().to_string();
287        }
288    }
289    // Optional bearer auth — `Some(token)` means the operator opted in.
290    // `get(..)` slicing: a hostile non-ASCII header can't panic the loop.
291    if let Some(token) = token {
292        let ok = authorization
293            .get(..7)
294            .is_some_and(|s| s.eq_ignore_ascii_case("bearer "))
295            && authorization.get(7..).is_some_and(|t| t == token);
296        if !ok {
297            return respond(stream, 401, b"unauthorized");
298        }
299    }
300    let Some(key) = sanitize_key(&raw_path) else {
301        return respond(stream, 400, b"bad path");
302    };
303    match method.as_str() {
304        "GET" => match fs::read(root.join(&key)) {
305            Ok(bytes) => respond(stream, 200, &bytes),
306            Err(_) => respond(stream, 404, b"not found"),
307        },
308        "PUT" => {
309            // Reject the declared length BEFORE allocating — a huge
310            // Content-Length must not become a huge vec.
311            if content_length > MAX_PUT {
312                return respond(stream, 413, b"payload too large");
313            }
314            let mut body = vec![0u8; content_length as usize];
315            reader
316                .read_exact(&mut body)
317                .map_err(Error::io("read body"))?;
318            let path = root.join(&key);
319            let parent = path.parent().unwrap_or(Path::new("."));
320            fs::create_dir_all(parent).map_err(Error::io("create blob dir"))?;
321            fs::write(&path, &body).map_err(Error::io(format!("write {}", path.display())))?;
322            respond(stream, 200, b"ok")
323        }
324        _ => respond(stream, 405, b"method not allowed"),
325    }
326}

328/// A request path → a safe relative key (no traversal; `/` → `index.json`).
329fn sanitize_key(raw: &str) -> Option<PathBuf> {
330    let trimmed = raw.split('?').next().unwrap_or(raw).trim_start_matches('/');
331    let mut rel = PathBuf::new();
332    for component in Path::new(trimmed).components() {
333        match component {
334            std::path::Component::Normal(p) => rel.push(p),
335            std::path::Component::CurDir => {}
336            _ => return None,
337        }
338    }
339    if rel.as_os_str().is_empty() {
340        rel.push("index.json");
341    }
342    Some(rel)
343}

345fn respond(stream: &mut std::net::TcpStream, code: u16, body: &[u8]) -> Result<(), Error> {
346    use std::io::Write;
347    let reason = match code {
348        200 => "OK",
349        400 => "Bad Request",
350        401 => "Unauthorized",
351        404 => "Not Found",
352        405 => "Method Not Allowed",
353        413 => "Payload Too Large",
354        _ => "Error",
355    };
356    let head = format!(
357        "HTTP/1.1 {code} {reason}\r\nContent-Length: {}\r\nContent-Type: application/octet-stream\r\nConnection: close\r\n\r\n",
358        body.len()
359    );
360    stream
361        .write_all(head.as_bytes())
362        .map_err(Error::io("write head"))?;
363    stream.write_all(body).map_err(Error::io("write body"))?;
364    stream.flush().map_err(Error::io("flush"))?;
365    Ok(())
366}
