3//! tamias-source-http — the tarball ref kind + source:
4//! `http(s)://….tgz|.tar.gz` GETs, verified against the optional `#sha256:`
5//! pin, unpacked into the fetch dir. The tarball's sha256 is the content key
6//! (`sha256-<hex>`), so a pinned ref + warm cache never touches the network —
7//! the air-gapped path.
8

16/// Largest tarball a fetch buffers — 512 MiB.
17const MAX_TARBALL: u64 = 512 * 1024 * 1024;
18
19/// Global request timeout — a hung host can't block a fetch forever.
20const TIMEOUT: std::time::Duration = std::time::Duration::from_secs(300);
21
22/// Connect timeout — fail fast on an unreachable host.
23const CONNECT_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(15);

25/// The http ref kind — a tarball URL with an optional digest pin.
26#[derive(Clone, Copy, Debug, Default)]
27pub struct HttpRefKind;
28
29/// A parsed http ref.
30#[derive(Clone, Debug, PartialEq, Eq)]
31pub struct HttpSpec {
32    /// URL minus the `#sha256:` fragment.
33    pub url: String,
34    /// Expected tarball sha256 hex, when pinned.
35    pub digest: Option<String>,
36}
37
38impl RefKind for HttpRefKind {
39    fn claims(&self, input: &str) -> bool {
40        input.starts_with("https://") || input.starts_with("http://")
41    }
42
43    /// URLs are case-sensitive — the extension check is deliberately exact.
44    #[allow(clippy::case_sensitive_file_extension_comparisons)]
45    fn parse(&self, input: &str) -> Result<Spec, Error> {
46        let (url, fragment) = match input.split_once('#') {
47            Some((url, fragment)) => (url, Some(fragment)),
48            None => (input, None),
49        };
50        let digest = match fragment {
51            None => None,
52            Some(fragment) => {
53                let hex = fragment
54                    .strip_prefix("sha256:")
55                    .or_else(|| fragment.strip_prefix("sha256="))
56                    .ok_or_else(|| {
57                        Error::Failed(format!(
58                            "unsupported fragment '#{fragment}' — expected '#sha256:<hex>'"
59                        ))
60                    })?;
61                if hex.len() != 64 || !hex.bytes().all(|b| b.is_ascii_hexdigit()) {
62                    return Err(Error::Failed(
63                        "sha256 digest must be 64 hex chars".to_string(),
64                    ));
65                }
66                Some(hex.to_ascii_lowercase())
67            }
68        };
69        if !(url.ends_with(".tgz") || url.ends_with(".tar.gz")) {
70            return Err(Error::Failed(format!(
71                "http refs must point at a .tgz/.tar.gz tarball: {url}"
72            )));
73        }
74        Ok(Box::new(HttpSpec {
75            url: url.to_string(),
76            digest,
77        }))
78    }
79}

81/// Whether a fetch is cleartext (`http://`) AND unpinned — the case that
82/// deserves the stderr warning.
83fn cleartext_unpinned(url: &str, digest: &Option<String>) -> bool {
84    url.starts_with("http://") && digest.is_none()
85}

87/// The http source.
88pub struct HttpSource {
89    agent: ureq::Agent,
90}

92impl HttpSource {
93    /// An http source over a default ureq agent.
94    #[must_use]
95    pub fn new() -> Self {
96        Self::with_agent(ureq::Agent::new_with_config(
97            ureq::Agent::config_builder()
98                .http_status_as_error(false)
99                .timeout_connect(Some(CONNECT_TIMEOUT))
100                .timeout_global(Some(TIMEOUT))
101                .build(),
102        ))
103    }
104
105    /// An http source over a custom agent (proxies, timeouts, fakes).
106    #[must_use]
107    pub fn with_agent(agent: ureq::Agent) -> Self {
108        Self { agent }
109    }
110}

112impl Default for HttpSource {

118/// An http fetch plan — reusable by other sources (e.g. `cf:`).
119pub struct HttpPlan {
120    url: String,
121    digest: Option<String>,
122}
123
124impl HttpPlan {
125    /// A plan for `url`, optionally digest-pinned.
126    #[must_use]
127    pub fn new(url: impl Into<String>, digest: Option<String>) -> Self {
128        Self {
129            url: url.into(),
130            digest,
131        }
132    }
133}
134
135impl Plan for HttpPlan {
136    fn prefetch_identity(&self) -> Option<(String, String)> {
137        self.digest
138            .as_ref()
139            .map(|d| (format!("sha256-{d}"), format!("sha256:{d}")))
140    }
141}
142
143impl Source for HttpSource {
144    fn accepts(&self, spec: &dyn Any) -> bool {
145        spec.is::<HttpSpec>()
146    }
147
148    fn plan(&self, spec: &dyn Any) -> Result<Box<dyn Plan>, Error> {
149        let HttpSpec { url, digest } = spec
150            .downcast_ref::<HttpSpec>()
151            .ok_or_else(|| Error::Failed("HttpSource only plans http refs".to_string()))?;
152        Ok(Box::new(HttpPlan {
153            url: url.clone(),
154            digest: digest.clone(),
155        }))
156    }
157
158    fn fetch(&self, plan: &dyn Plan, dest_dir: &Path) -> Result<Fetched, Error> {
159        let HttpPlan { url, digest } = plan
160            .downcast_ref::<HttpPlan>()
161            .ok_or_else(|| Error::Failed("HttpSource only fetches http plans".to_string()))?;
162        // Cleartext + unauthenticated + (usually) unpinned — worth a loud
163        // stderr note; `--json` stdout stays clean. Warn at fetch time:
164        // parse/plan also run in library contexts where eprintln is rude.
165        if cleartext_unpinned(url, digest) {
166            eprintln!(
167                "warning: fetching {url} over plain http with no #sha256: pin — \
168                 the payload is unauthenticated and modifiable in transit"
169            );
170        }
171        fs::create_dir_all(dest_dir)
172            .map_err(Error::io(format!("create {}", dest_dir.display())))?;
173        let mut response = self
174            .agent
175            .get(url)
176            .call()
177            .map_err(|e| Error::Failed(format!("GET {url}: {e}")))?;
178        if !response.status().is_success() {
179            return Err(Error::Failed(format!(
180                "GET {url}: HTTP {}",
181                response.status()
182            )));
183        }
184        let bytes = response
185            .body_mut()
186            .with_config()
187            .limit(MAX_TARBALL)
188            .read_to_vec()
189            .map_err(|e| Error::Failed(format!("read {url}: {e}")))?;
190        let hex = sha256_hex(&bytes);
191        if let Some(want) = digest.as_deref().filter(|want| hex.as_str() != *want) {
192            return Err(Error::Failed(format!(
193                "digest mismatch for {url}: sha256:{hex} ≠ pinned sha256:{want}"
194            )));
195        }
196        untar_gz(bytes.as_slice(), dest_dir)?;
197        Ok(Fetched {
198            dir: dest_dir.to_path_buf(),
199            resolved: format!("sha256:{hex}"),
200            content_key: format!("sha256-{hex}"),
201        })
202    }
203}

219    /// Serve `body` once with `status` on a random port.
220    fn serve(body: Vec<u8>, status: &str) -> u16 {
221        let listener = TcpListener::bind("127.0.0.1:0").unwrap();
222        let port = listener.local_addr().unwrap().port();
223        let status = status.to_string();
224        std::thread::spawn(move || {
225            if let Ok((mut s, _)) = listener.accept() {
226                let mut buf = [0u8; 4096];
227                let _ = s.read(&mut buf);
228                let head = format!(
229                    "HTTP/1.1 {status}\r\nContent-Length: {}\r\nConnection: close\r\n\r\n",
230                    body.len()
231                );
232                let _ = s.write_all(head.as_bytes());
233                let _ = s.write_all(&body);
234                let _ = s.flush();
235            }
236        });
237        port
238    }

240    fn tarball() -> Vec<u8> {
241        let src = tempfile::tempdir().unwrap();
242        std::fs::write(src.path().join("a.txt"), "a").unwrap();
243        let out = tempfile::tempdir().unwrap();
244        let tgz = out.path().join("x.tgz");
245        pack_tgz(src.path(), &tgz).unwrap();
246        std::fs::read(&tgz).unwrap()
247    }

249    #[test]
250    fn parse_plan_fetch_and_errors() {
251        let refs = Refs::new().with(Arc::new(HttpRefKind));
252        let sources = Sources::new().with(Arc::new(HttpSource::new()));
253        let bytes = tarball();
254        let hex = sha256_hex(&bytes);
255
256        // unpinned happy path
257        let port = serve(bytes.clone(), "200 OK");
258        let spec = refs
259            .parse(&format!("http://127.0.0.1:{port}/x.tgz"))
260            .unwrap();
261        let planned = sources.plan(&*spec).unwrap();
262        assert!(planned.prefetch_identity().is_none());
263        let dest = tempfile::tempdir().unwrap();
264        assert!(planned.fetch(dest.path()).is_ok());
265        assert!(dest.path().join("a.txt").is_file());
266
267        // digest-pinned → prefetch identity + verified
268        let port = serve(bytes.clone(), "200 OK");
269        let spec = refs
270            .parse(&format!("http://127.0.0.1:{port}/x.tgz#sha256:{hex}"))
271            .unwrap();
272        let planned = sources.plan(&*spec).unwrap();
273        assert_eq!(
274            planned.prefetch_identity().unwrap().0,
275            format!("sha256-{hex}")
276        );
277        let d2 = tempfile::tempdir().unwrap();
278        assert!(planned.fetch(d2.path()).is_ok());
279
280        // grammar errors + claims
281        assert!(refs.parse("http://x/y.zip").is_err());
282        assert!(refs.parse("http://x/y.tgz#nope:1").is_err());
283        assert!(refs
284            .parse(&format!("http://x/y.tgz#sha256:{}", "z".repeat(64)))
285            .is_err());
286        assert!(HttpRefKind.claims("https://x/y.tgz"));
287        assert!(!HttpRefKind.claims("git:x"));
288
289        // HTTP error status
290        let port = serve(Vec::new(), "500 X");
291        let spec = refs
292            .parse(&format!("http://127.0.0.1:{port}/x.tgz"))
293            .unwrap();
294        let planned = sources.plan(&*spec).unwrap();
295        let d3 = tempfile::tempdir().unwrap();
296        assert!(planned.fetch(d3.path()).is_err());
297
298        // digest mismatch
299        let port = serve(bytes.clone(), "200 OK");
300        let spec = refs
301            .parse(&format!(
302                "http://127.0.0.1:{port}/x.tgz#sha256:{}",
303                "a".repeat(64)
304            ))
305            .unwrap();
306        let planned = sources.plan(&*spec).unwrap();
307        let d4 = tempfile::tempdir().unwrap();
308        assert!(planned.fetch(d4.path()).is_err());
309
310        // dead server
311        let spec = refs.parse("http://127.0.0.1:1/x.tgz").unwrap();
312        let planned = sources.plan(&*spec).unwrap();
313        let d5 = tempfile::tempdir().unwrap();
314        assert!(planned.fetch(d5.path()).is_err());
315
316        // foreign spec / plan
317        let source = HttpSource::new();
318        assert!(!source.accepts(&1u8));
319        assert!(source.plan(&1u8).is_err());
320        assert!(source.fetch(&ForeignPlan, d5.path()).is_err());
321
322        // the public plan builder — reusable by other sources (`cf:`)
323        let port = serve(bytes.clone(), "200 OK");
324        let built = HttpPlan::new(format!("http://127.0.0.1:{port}/x.tgz"), None);
325        let d6 = tempfile::tempdir().unwrap();
326        assert!(source.fetch(&built, d6.path()).is_ok());
327        assert!(HttpPlan::new("http://x/y.tgz", None)
328            .prefetch_identity()
329            .is_none());
330    }
